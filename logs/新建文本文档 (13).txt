不对。不是"之上"，是之下。

它们不是条件概率关系。九肖和六肖是同一个池子的两个不同输出：

    ↓

    ↓
排序取Top6 ← 这就是六肖（92.02%）
六肖是九肖的子集。六肖命中必须九肖先命中——如果九肖没命中（答案不在九肖候选里），六肖也不可能命中。所以六肖的92.02%是九肖93.49%减去排序损失30次得到的。

正确的关系：六肖命中率 = 九肖命中率 - 排序损失率。不是叠加上去的，是缩水下来的。

那么六肖的概率在相对总体概率来说是不是实际上不到92%,因为九肖只有93%

概率关系
九肖命中率：93.49%（1981期中1852期命中）

六肖命中率：92.02%（1981期中1823期命中）

排序损失：1852 - 1823 = 29期（九肖命中但六肖未命中）

92.02%是实打实的，不依附于九肖。 它不是"九肖命中后的六肖命中率"（条件概率），而是"所有期中六肖直接命中的比例"（绝对概率）。

条件概率的话：在九肖命中的1852期里，六肖命中1823期，条件概率是1823/1852=98.4%。但实战中你不知道当期九肖是否命中，所以你面对的就是绝对概率——92.02%。

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
core_update_rules.py — 特肖动态杀肖规则库 更新脚本
============================================================
用途：每次数据更新后运行，重新提取≥95%+连错≤1的规则
用法：
  python core_update_rules.py
  → 输出 d:\lottery_ai\特肖杀肖规则库.json
============================================================
"""
import json, os, sys
from collections import defaultdict, Counter

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
MARK6_DIR = os.path.join(BASE_DIR, "mark6") if os.path.exists(os.path.join(BASE_DIR, "mark6")) else BASE_DIR
if MARK6_DIR not in sys.path: sys.path.insert(0, MARK6_DIR)

from shuju_loader import load_all_data
from shx_suishu import get_shengxiao_by_suima, SHENGXIAO

ZODIAC = SHENGXIAO
POS_NAMES = ["平一", "平二", "平三", "平四", "平五", "平六", "特码"]
OUTPUT_PATH = os.path.join(BASE_DIR, "特肖杀肖规则库.json")

MIN_SAMPLES = 5       # 最少样本数
MIN_KILL_RATE = 95.0  # 最低杀对率
MAX_STREAK = 1        # 最大允许连错


def offset_num(num, offset):
    return (num - 1 + offset) % 49 + 1


def build_rules(records, min_samples=MIN_SAMPLES, min_rate=MIN_KILL_RATE, max_streak=MAX_STREAK):
    """
    从全部历史数据中提取规则库
    返回: {特肖: {位置: {触发平码生肖: [(偏移,杀肖,杀对率,样本量,连错),...]}}}
    """
    # 统计每种(特肖, 位置, 偏移, 触发平码生肖) → 所得生肖的杀对率和连错
    stats = defaultdict(lambda: defaultdict(lambda: {"total": 0, "hit": 0, "result_sx": None}))

    for i in range(len(records) - 1):
        curr, nxt = records[i], records[i + 1]
        curr_sx = curr["te_sx"]
        year = curr["year"]
        nxt_sx = nxt["te_sx"]

        for pos_idx, pos_name in enumerate(POS_NAMES):
            num = curr["ping_nums"][pos_idx] if pos_idx < 6 else curr["te_num"]
            trigger_sx = get_shengxiao_by_suima(num, year)

            for off in range(12):
                new_num = offset_num(num, off)
                result_sx = get_shengxiao_by_suima(new_num, year)
                key = (pos_name, off, trigger_sx)
                stats[curr_sx][key]["total"] += 1
                stats[curr_sx][key]["result_sx"] = result_sx
                if result_sx != nxt_sx:
                    stats[curr_sx][key]["hit"] += 1

    # 筛选并计算连错
    rules = {}
    for sx in ZODIAC:
        if sx not in stats:
            continue
        rules[sx] = {}
        for (pos_name, off, trigger_sx), v in stats[sx].items():
            if v["total"] < min_samples:
                continue
            kill_rate = v["hit"] / v["total"] * 100
            if kill_rate < min_rate:
                continue

            # 计算连错（遍历历史查找该规则触发时的连续错误）
            max_streak_found = 0
            cur_streak = 0
            for i in range(len(records) - 1):
                if records[i]["te_sx"] != sx:
                    continue
                year_i = records[i]["year"]
                pos_idx = POS_NAMES.index(pos_name)
                num = records[i]["ping_nums"][pos_idx] if pos_idx < 6 else records[i]["te_num"]
                actual_sx = get_shengxiao_by_suima(num, year_i)
                if actual_sx != trigger_sx:
                    continue

                killed_sx = get_shengxiao_by_suima(offset_num(num, off), year_i)
                if records[i + 1]["te_sx"] == killed_sx:
                    cur_streak += 1
                    max_streak_found = max(max_streak_found, cur_streak)
                else:
                    cur_streak = 0

            if max_streak_found > max_streak:
                continue

            if pos_name not in rules[sx]:
                rules[sx][pos_name] = {}
            if trigger_sx not in rules[sx][pos_name]:
                rules[sx][pos_name][trigger_sx] = []

            rules[sx][pos_name][trigger_sx].append(
                (off, v["result_sx"], round(kill_rate, 2), v["total"], max_streak_found)
            )

    # 排序：同位置同触发内按杀对率降序
    for sx in rules:
        for pos in rules[sx]:
            for tri in rules[sx][pos]:
                rules[sx][pos][tri].sort(key=lambda x: x[2], reverse=True)

    return rules


def main():
    print("=" * 55)
    print("core_update_rules.py — 规则库更新")
    print("=" * 55)

    print("加载数据...")
    data = load_all_data(auto_update=False)
    records = []
    for item in data:
        try:
            qs = str(item.get("expect", ""))
            oc = str(item.get("openCode", ""))
            ot = item.get("openTime", "")
            year = int(ot[:4]) if ot else (int(qs[:4]) if len(qs) >= 4 else 2026)
            if not qs or not oc: continue
            parts = oc.strip().split(",")
            if len(parts) != 7: continue
            nums = [int(p.strip()) for p in parts]
            records.append({
                "qishu": qs, "year": year,
                "te_num": nums[6], "te_sx": get_shengxiao_by_suima(nums[6], year),
                "te_wei": nums[6] % 10,
                "ping_nums": nums[:6],
            })
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    print(f"记录数: {len(records)}")

    print("提取规则...")
    rules = build_rules(records)

    # 备份旧文件
    if os.path.exists(OUTPUT_PATH):
        backup = OUTPUT_PATH + ".bak"
        os.replace(OUTPUT_PATH, backup)
        print(f"已备份旧规则库 → {backup}")

    # 保存
    with open(OUTPUT_PATH, 'w', encoding='utf-8') as f:
        json.dump(rules, f, ensure_ascii=False, indent=2)

    print(f"✅ 已保存: {OUTPUT_PATH}")
    total_rules = sum(
        len(rules[sx][pos][tri])
        for sx in rules for pos in rules[sx] for tri in rules[sx][pos]
    )
    print(f"   {len(rules)}个生肖, {total_rules}条规则")
    for sx in rules:
        cnt = sum(len(rules[sx][p][t]) for p in rules[sx] for t in rules[sx][p])
        print(f"   {sx}: {len(rules[sx])}个位置, {cnt}条规则")

    print(f"\n===== 更新完成 =====")


if __name__ == "__main__":
    main()

ZODIAC = SHENGXIAO
POS_NAMES = ["平一", "平二", "平三", "平四", "平五", "平六", "特码"]

MIN_SAMPLES = 5       # 最少样本数
MIN_KILL_RATE = 95.0  # 最低杀对率
MAX_STREAK = 1        # 最大允许连错


def offset_num(num, offset):
    return (num - 1 + offset) % 49 + 1


def build_rules(records, min_samples=MIN_SAMPLES, min_rate=MIN_KILL_RATE, max_streak=MAX_STREAK):
    """
    从全部历史数据中提取规则库
    返回: {特肖: {位置: {触发平码生肖: [(偏移,杀肖,杀对率,样本量,连错),...]}}}
    """
    # 统计每种(特肖, 位置, 偏移, 触发平码生肖) → 所得生肖的杀对率和连错
    stats = defaultdict(lambda: defaultdict(lambda: {"total": 0, "hit": 0, "result_sx": None}))

    for i in range(len(records) - 1):
        curr, nxt = records[i], records[i + 1]
        curr_sx = curr["te_sx"]
        year = curr["year"]
        nxt_sx = nxt["te_sx"]

        for pos_idx, pos_name in enumerate(POS_NAMES):
            num = curr["ping_nums"][pos_idx] if pos_idx < 6 else curr["te_num"]
            trigger_sx = get_shengxiao_by_suima(num, year)

            for off in range(12):
                new_num = offset_num(num, off)
                result_sx = get_shengxiao_by_suima(new_num, year)
                key = (pos_name, off, trigger_sx)
                stats[curr_sx][key]["total"] += 1
                stats[curr_sx][key]["result_sx"] = result_sx
                if result_sx != nxt_sx:
                    stats[curr_sx][key]["hit"] += 1

    # 筛选并计算连错
    rules = {}
    for sx in ZODIAC:
        if sx not in stats:
            continue
        rules[sx] = {}
        for (pos_name, off, trigger_sx), v in stats[sx].items():
            if v["total"] < min_samples:
                continue
            kill_rate = v["hit"] / v["total"] * 100
            if kill_rate < min_rate:
                continue

            # 计算连错（遍历历史查找该规则触发时的连续错误）
            max_streak_found = 0
            cur_streak = 0
            for i in range(len(records) - 1):
                if records[i]["te_sx"] != sx:
                    continue
                year_i = records[i]["year"]
                pos_idx = POS_NAMES.index(pos_name)
                num = records[i]["ping_nums"][pos_idx] if pos_idx < 6 else records[i]["te_num"]
                actual_sx = get_shengxiao_by_suima(num, year_i)
                if actual_sx != trigger_sx:
                    continue

                killed_sx = get_shengxiao_by_suima(offset_num(num, off), year_i)
                if records[i + 1]["te_sx"] == killed_sx:
                    cur_streak += 1
                    max_streak_found = max(max_streak_found, cur_streak)
                else:
                    cur_streak = 0

            if max_streak_found > max_streak:
                continue

            if pos_name not in rules[sx]:
                rules[sx][pos_name] = {}
            if trigger_sx not in rules[sx][pos_name]:
                rules[sx][pos_name][trigger_sx] = []

            rules[sx][pos_name][trigger_sx].append(
                (off, v["result_sx"], round(kill_rate, 2), v["total"], max_streak_found)
            )

    # 排序：同位置同触发内按杀对率降序
    for sx in rules:
        for pos in rules[sx]:
            for tri in rules[sx][pos]:
                rules[sx][pos][tri].sort(key=lambda x: x[2], reverse=True)

    return rules

            actual_sx = records[up_to - 1]["ping_sx"][pos_idx] if pos_idx < 6 else cur_sx
            if actual_sx in SHAXIAO_RULES[cur_sx][pos_name]:
                rules = SHAXIAO_RULES[cur_sx][pos_name][actual_sx]
                kills.add(rules[0][1])
                for r in rules:
                    rule_scores[r[1]] -= r[2]

    # 2. 剩余+补足9肖
    remaining = [s for s in ZODIAC if s not in kills]
    if len(remaining) < 9:
        freq = Counter(r["te_sx"] for r in records[:up_to])
        for s, _ in freq.most_common(12):
            if s not in kills and s not in remaining:
                remaining.append(s)
            if len(remaining) >= 9: break
    nine = remaining[:9]

    freq = Counter(r["te_sx"] for r in records[:up_to])
    max_f = max(freq.values()) if freq else 1

    # 3. 规则库排序取Top6
    final_rule = Counter()
    for s in nine:
        fs = freq.get(s, 0) / max_f * 100 * 0.3
        rs = (rule_scores.get(s, 0) + 100) * 0.7
        final_rule[s] = fs + rs
    rule6 = set(s for s, _ in sorted(final_rule.items(), key=lambda x: x[1], reverse=True)[:6])

    # 4. 四模池内排序取Top6
    ta = defaultdict(Counter)
    tb = defaultdict(lambda: defaultdict(float))
    tc = defaultdict(Counter)
    td = defaultdict(Counter)
    for i in range(up_to - 1):
        ta[records[i]["te_sx"]][records[i + 1]["te_sx"]] += 1
    for i in range(up_to - 1):
        dist = up_to - 1 - i
        w = 1.5 if dist <= 12 else (1.2 if dist <= 24 else 1.0)
        tb[records[i]["te_sx"]][records[i + 1]["te_sx"]] += w
    for i in range(up_to - 2):
        tc[(records[i]["te_sx"], records[i + 1]["te_sx"])][records[i + 2]["te_sx"]] += 1
    for i in range(up_to - 1):
        td[records[i]["te_num"]][records[i + 1]["te_sx"]] += 1

    prev_sx = records[up_to - 2]["te_sx"] if up_to >= 2 else cur_sx
    cur_num = records[up_to - 1]["te_num"]
    model_scores = Counter()
    t = sum(ta[cur_sx].values())
    if t:
        for s, c in ta[cur_sx].items(): model_scores[s] += (c / t) * 0.30
    t = sum(tb[cur_sx].values())
    if t:
        for s, c in tb[cur_sx].items(): model_scores[s] += (c / t) * 0.30
    key = (prev_sx, cur_sx)
    if key in tc:
        t = sum(tc[key].values())
        if t:
            for s, c in tc[key].items(): model_scores[s] += (c / t) * 0.20
    t = sum(td[cur_num].values())
    if t:
        for s, c in td[cur_num].items(): model_scores[s] += (c / t) * 0.20

    if cur_sx in model_scores:
        del model_scores[cur_sx]
    if len(records[up_to - 1]["ping_nums"]) >= 2:
        p2 = records[up_to - 1]["ping_nums"][1]
        kill_m = get_shengxiao_by_suima(offset_num(p2, -9), cur_year)
        if kill_m in model_scores:
            del model_scores[kill_m]

    pool_scores = {s: model_scores.get(s, 0) for s in nine}
    model6 = set(s for s, _ in sorted(pool_scores.items(), key=lambda x: x[1], reverse=True)[:6])

    # 5. 并集
    union = list(rule6 | model6)
    return union, kills, nine, rule6, model6
