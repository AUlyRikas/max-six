#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
xiao30.py — 两条杀肖规则在连错期的贡献拆解
分析：每条规则单独误杀、同时误杀、与投票未覆盖的交叉
"""
import os, sys
from collections import defaultdict, Counter

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
MARK6_DIR = os.path.join(BASE_DIR, "mark6") if os.path.exists(os.path.join(BASE_DIR, "mark6")) else BASE_DIR
if MARK6_DIR not in sys.path: sys.path.insert(0, MARK6_DIR)

from shuju_loader import load_all_data
from shx_suishu import get_shengxiao_by_suima, SHENGXIAO

ZODIAC = SHENGXIAO

def offset_num(num, offset):
    return (num-1+offset)%49+1

def extract(data):
    records = []
    for item in data:
        try:
            qs=str(item.get("expect","")); oc=str(item.get("openCode",""))
            ot=item.get("openTime","")
            year=int(ot[:4]) if ot else (int(qs[:4]) if len(qs)>=4 else 2026)
            if not qs or not oc: continue
            parts=oc.strip().split(",")
            if len(parts)!=7: continue
            nums=[int(p.strip()) for p in parts]
            records.append({"qishu":qs,"year":year,"te_num":nums[6],"te_sx":get_shengxiao_by_suima(nums[6],year),"te_wei":nums[6]%10,"ping_nums":nums[:6],"all_nums":nums})
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records

def method_B_components(records, up_to):
    """返回四个关键信息：投票是否覆盖、两条杀肖各是否误杀"""
    if up_to<50: return None
    ta=defaultdict(Counter); tb=defaultdict(lambda:defaultdict(float)); tc=defaultdict(Counter); td=defaultdict(Counter)
    for i in range(up_to-1): ta[records[i]["te_sx"]][records[i+1]["te_sx"]]+=1
    for i in range(up_to-1):
        dist=up_to-1-i; w=1.5 if dist<=12 else (1.2 if dist<=24 else 1.0)
        tb[records[i]["te_sx"]][records[i+1]["te_sx"]]+=w
    for i in range(up_to-2): tc[(records[i]["te_sx"],records[i+1]["te_sx"])][records[i+2]["te_sx"]]+=1
    for i in range(up_to-1): td[records[i]["te_num"]][records[i+1]["te_sx"]]+=1
    cur_sx=records[up_to-1]["te_sx"]; prev_sx=records[up_to-2]["te_sx"] if up_to>=2 else cur_sx
    cur_num=records[up_to-1]["te_num"]; cur_year=records[up_to-1]["year"]
    actual=records[up_to]["te_sx"] if up_to<len(records) else None

    scores=Counter()
    t=sum(ta[cur_sx].values())
    if t: 
        for s,c in ta[cur_sx].items(): scores[s]+=(c/t)*0.30
    t=sum(tb[cur_sx].values())
    if t: 
        for s,c in tb[cur_sx].items(): scores[s]+=(c/t)*0.30
    t=sum(tc[(prev_sx,cur_sx)].values()) if (prev_sx,cur_sx) in tc else 0
    if t: 
        for s,c in tc[(prev_sx,cur_sx)].items(): scores[s]+=(c/t)*0.20
    t=sum(td[cur_num].values())
    if t: 
        for s,c in td[cur_num].items(): scores[s]+=(c/t)*0.20

    # 投票是否覆盖（杀肖前看实际特肖在不在Top9）
    vote_top9_before_kill = set(s for s,_ in scores.most_common(9))
    vote_covered = actual in vote_top9_before_kill if actual else None

    # 杀1
    kill_self_wrong = (cur_sx == actual)
    if cur_sx in scores: del scores[cur_sx]

    # 杀2
    kill_num9 = None
    kill_num9_wrong = False
    if len(records[up_to-1]["ping_nums"])>=2:
        p2=records[up_to-1]["ping_nums"][1]
        kill_num9=get_shengxiao_by_suima(offset_num(p2,-9),cur_year)
        kill_num9_wrong = (kill_num9 == actual)
        if kill_num9 in scores: del scores[kill_num9]

    return {
        "vote_covered": vote_covered,
        "kill_self_wrong": kill_self_wrong,
        "kill_num9_wrong": kill_num9_wrong,
        "kill_self_sx": cur_sx,
        "kill_num9_sx": kill_num9,
        "actual": actual,
    }

def main():
    print("xiao30.py — 杀肖规则贡献拆解")
    print("="*65)
    data=load_all_data(auto_update=False); records=extract(data)
    min_hist=200

    # 全局统计 + 连错期统计
    all_results = []
    miss_results = []

    for idx in range(min_hist, len(records)):
        comp = method_B_components(records, idx)
        if comp is None: continue
        all_results.append(comp)
        # 判断最终是否命中
        final_hit = True
        if not comp["vote_covered"]: final_hit = False
        elif comp["kill_self_wrong"]: final_hit = False
        elif comp["kill_num9_wrong"]: final_hit = False
        if not final_hit:
            miss_results.append(comp)

    total_all = len(all_results)
    total_miss = len(miss_results)

    print(f"\n总期数: {total_all}  |  未命中: {total_miss} ({total_miss/total_all*100:.1f}%)")

    # 全局交叉统计
    print(f"\n{'='*65}")
    print("全局交叉分析（所有期数）:")
    print(f"{'='*65}")

    vote_cov = sum(1 for r in all_results if r["vote_covered"])
    vote_not = sum(1 for r in all_results if not r["vote_covered"])
    self_ok = sum(1 for r in all_results if not r["kill_self_wrong"])
    self_bad = sum(1 for r in all_results if r["kill_self_wrong"])
    num9_ok = sum(1 for r in all_results if not r["kill_num9_wrong"])
    num9_bad = sum(1 for r in all_results if r["kill_num9_wrong"])

    print(f"投票覆盖: {vote_cov} ({vote_cov/total_all*100:.1f}%) | 未覆盖: {vote_not} ({vote_not/total_all*100:.1f}%)")
    print(f"杀自正确: {self_ok} ({self_ok/total_all*100:.1f}%) | 误杀自: {self_bad} ({self_bad/total_all*100:.1f}%)")
    print(f"num9正确: {num9_ok} ({num9_ok/total_all*100:.1f}%) | 误杀num9: {num9_bad} ({num9_bad/total_all*100:.1f}%)")

    # 全部错期里的贡献
    print(f"\n{'='*65}")
    print(f"未命中{total_miss}期的原因拆解:")
    print(f"{'='*65}")

    only_vote = sum(1 for r in miss_results if not r["vote_covered"] and not r["kill_self_wrong"] and not r["kill_num9_wrong"])
    only_self = sum(1 for r in miss_results if r["vote_covered"] and r["kill_self_wrong"] and not r["kill_num9_wrong"])
    only_num9 = sum(1 for r in miss_results if r["vote_covered"] and not r["kill_self_wrong"] and r["kill_num9_wrong"])
    vote_and_self = sum(1 for r in miss_results if not r["vote_covered"] and r["kill_self_wrong"] and not r["kill_num9_wrong"])
    vote_and_num9 = sum(1 for r in miss_results if not r["vote_covered"] and not r["kill_self_wrong"] and r["kill_num9_wrong"])
    both_kill = sum(1 for r in miss_results if r["vote_covered"] and r["kill_self_wrong"] and r["kill_num9_wrong"])
    all_three = sum(1 for r in miss_results if not r["vote_covered"] and r["kill_self_wrong"] and r["kill_num9_wrong"])

    print(f"  仅投票未覆盖:         {only_vote:4d}次 ({only_vote/total_miss*100:5.1f}%)")
    print(f"  仅杀自误杀:           {only_self:4d}次 ({only_self/total_miss*100:5.1f}%)")
    print(f"  仅num9误杀:           {only_num9:4d}次 ({only_num9/total_miss*100:5.1f}%)")
    print(f"  投票未覆盖+杀自误杀:    {vote_and_self:4d}次 ({vote_and_self/total_miss*100:5.1f}%)")
    print(f"  投票未覆盖+num9误杀:   {vote_and_num9:4d}次 ({vote_and_num9/total_miss*100:5.1f}%)")
    print(f"  杀自+n9同时误杀(投票覆盖): {both_kill:4d}次 ({both_kill/total_miss*100:5.1f}%)")
    print(f"  三者全错:             {all_three:4d}次 ({all_three/total_miss*100:5.1f}%)")

    # 最关键的：哪些是"原本投票对的，被误杀搞错的"
    vote_right_killed = sum(1 for r in miss_results if r["vote_covered"] and (r["kill_self_wrong"] or r["kill_num9_wrong"]))
    print(f"\n  投票本来正确、被误杀弄错: {vote_right_killed}次 ({vote_right_killed/total_miss*100:.1f}%)")
    print(f"  投票本来就不对、杀不杀都错: {total_miss - vote_right_killed}次 ({(total_miss-vote_right_killed)/total_miss*100:.1f}%)")

    # 两条杀肖规则的错误重叠度
    both_wrong = sum(1 for r in all_results if r["kill_self_wrong"] and r["kill_num9_wrong"])
    print(f"\n两条杀肖同时误杀: {both_wrong}次 ({both_wrong/total_all*100:.2f}%)")

    # 同生肖被两条规则杀
    same_target = sum(1 for r in all_results if r["kill_self_sx"] == r["kill_num9_sx"])
    print(f"两规则杀同一生肖: {same_target}次 ({same_target/total_all*100:.1f}%)")

    print(f"\n===== xiao30.py 完成 =====")

if __name__=="__main__":
    main()

def predict_reverse(records, up_to, kill_mode="none"):
    """高危期：反向投票（取得分最低的9肖）+ 杀肖"""
    scores = get_scores(records, up_to)
    cur_sx=records[up_to-1]["te_sx"]; cur_year=records[up_to-1]["year"]
    
    # 被杀生肖永远排除
    killed_set = set()
    if kill_mode in ("both","self"):
        killed_set.add(cur_sx)
        if cur_sx in scores: del scores[cur_sx]
    if kill_mode == "both":
        if len(records[up_to-1]["ping_nums"])>=2:
            p2=records[up_to-1]["ping_nums"][1]
            kill=get_shengxiao_by_suima(offset_num(p2,-9),cur_year)
            killed_set.add(kill)
            if kill in scores: del scores[kill]
    
    # 反向：取最低分9个（排除已被杀的）
    bottom9 = [s for s,_ in scores.most_common()[-9:]]
    result = set(bottom9)
    # 被杀生肖加回来
    result |= killed_set
    # 如果不够9个，补高分
    if len(result) < 9:
        top = [s for s,_ in scores.most_common(9)]
        for s in top:
            result.add(s)
            if len(result) >= 9: break
    return set(list(result)[:9])

def backtest_streak(name, results_list):
    hits=sum(results_list); total=len(results_list)
    rate=hits/total*100 if total>0 else 0
    max_streak=cur=0; streak_dist=Counter()
    for r in results_list:
        if not r: cur+=1
        else:
            if cur>0: streak_dist[cur]+=1
            cur=0
    if cur>0: streak_dist[cur]+=1
    if streak_dist: max_streak=max(streak_dist.keys())
    over2=sum(v for k,v in streak_dist.items() if k>=3)
    print(f"  {name}: {rate:.2f}% 最大连错{max_streak}期 连错≥3:{over2}次")
    return rate

            records.append({"qishu":qs,"year":year,"te_num":nums[6],"te_sx":get_shengxiao_by_suima(nums[6],year),"te_wei":nums[6]%10,"ping_nums":nums[:6]})
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records

def predict_with_model_tweak(records, up_to, reverse_models=None, boost_models=None, kill_mode="both"):
    """
    reverse_models: set of model names to reverse (take bottom instead of top)
    boost_models: dict of model->weight multiplier
    """
    ta=defaultdict(Counter); tb=defaultdict(lambda:defaultdict(float)); tc=defaultdict(Counter); td=defaultdict(Counter)
    for i in range(up_to-1): ta[records[i]["te_sx"]][records[i+1]["te_sx"]]+=1
    for i in range(up_to-1):
        dist=up_to-1-i; w=1.5 if dist<=12 else (1.2 if dist<=24 else 1.0)
        tb[records[i]["te_sx"]][records[i+1]["te_sx"]]+=w
    for i in range(up_to-2): tc[(records[i]["te_sx"],records[i+1]["te_sx"])][records[i+2]["te_sx"]]+=1
    for i in range(up_to-1): td[records[i]["te_num"]][records[i+1]["te_sx"]]+=1
    cur_sx=records[up_to-1]["te_sx"]; prev_sx=records[up_to-2]["te_sx"] if up_to>=2 else cur_sx
    cur_num=records[up_to-1]["te_num"]; cur_year=records[up_to-1]["year"]

    scores=Counter()
    reverse_scores=Counter()

    # 模型A
    t=sum(ta[cur_sx].values())
    if t:
        for s,c in ta[cur_sx].items():
            if "A" in (reverse_models or set()):
                reverse_scores[s] += (c/t)*0.30
            else:
                scores[s] += (c/t)*0.30 * (boost_models.get("A",1.0) if boost_models else 1.0)
    # 模型B
    t=sum(tb[cur_sx].values())
    if t:
        for s,c in tb[cur_sx].items():
            if "B" in (reverse_models or set()):
                reverse_scores[s] += (c/t)*0.30
            else:
                scores[s] += (c/t)*0.30 * (boost_models.get("B",1.0) if boost_models else 1.0)
    # 模型C
    t=sum(tc[(prev_sx,cur_sx)].values()) if (prev_sx,cur_sx) in tc else 0
    if t:
        for s,c in tc[(prev_sx,cur_sx)].items():
            if "C" in (reverse_models or set()):
                reverse_scores[s] += (c/t)*0.20
            else:
                scores[s] += (c/t)*0.20 * (boost_models.get("C",1.0) if boost_models else 1.0)
    # 模型D
    t=sum(td[cur_num].values())
    if t:
        for s,c in td[cur_num].items():
            if "D" in (reverse_models or set()):
                reverse_scores[s] += (c/t)*0.20
            else:
                scores[s] += (c/t)*0.20 * (boost_models.get("D",1.0) if boost_models else 1.0)

    # 合并：正向得分 - 反向得分
    final = Counter()
    for s in ZODIAC:
        final[s] = scores.get(s,0) - reverse_scores.get(s,0)

    # 杀肖
    if kill_mode in ("both","self"):
        if cur_sx in final: del final[cur_sx]
    if kill_mode=="both":
        if len(records[up_to-1]["ping_nums"])>=2:
            p2=records[up_to-1]["ping_nums"][1]
            kill=get_shengxiao_by_suima(offset_num(p2,-9),cur_year)
            if kill in final: del final[kill]

    top9=[s for s,_ in final.most_common(9)]
    if len(top9)<9:
        freq=Counter(r["te_sx"] for r in records[:up_to])
        for s,_ in freq.most_common(12):
            if s not in top9: top9.append(s)
            if len(top9)>=9: break
    return set(top9[:9])

def backtest_streak(name, results_list):
    hits=sum(results_list); total=len(results_list)
    rate=hits/total*100 if total>0 else 0
    max_streak=cur=0; streak_dist=Counter()
    for r in results_list:
        if not r: cur+=1
        else:
            if cur>0: streak_dist[cur]+=1
            cur=0
    if cur>0: streak_dist[cur]+=1
    if streak_dist: max_streak=max(streak_dist.keys())
    over2=sum(v for k,v in streak_dist.items() if k>=3)
    print(f"  {name}: {rate:.2f}% 最大连错{max_streak}期 连错≥3:{over2}次")
    return rate
