from shuju_loader import load_all_data
from shx_suishu import get_shengxiao_by_suima, SHENGXIAO

# 生肖数组（来自shx_suishu模块）
ZODIAC = SHENGXIAO  # ['马','蛇','龙','兔','虎','牛','鼠','猪','狗','鸡','猴','羊']


# ==================== 基础工具函数 ====================

def offset_num(num, offset):
    """号码偏移，模49，范围1~49"""
    return (num - 1 + offset) % 49 + 1


def sx_window(sx, window=4):
    """
    以某生肖为中心 ±window → 返回生肖集合
    例如: sx_window('猪', 4) → 猪±4 = 9个生肖
    """
    idx = ZODIAC.index(sx)
    return set(ZODIAC[(idx + w) % 12] for w in range(-window, window + 1))


def extract_records(data):
    """
    从load_all_data返回的数据中提取所需字段
    返回按时间排序的记录列表
    """
    records = []
    for item in data:
        try:
            qs = str(item.get("expect", ""))
            oc = str(item.get("openCode", ""))
            ot = item.get("openTime", "")
            year = int(ot[:4]) if ot else (int(qs[:4]) if len(qs) >= 4 else 2026)
            if not qs or not oc:
                continue
            parts = oc.strip().split(",")
            if len(parts) != 7:
                continue
            nums = [int(p.strip()) for p in parts]
            records.append({
                "qishu": qs,
                "year": year,
                "te_num": nums[6],
                "te_sx": get_shengxiao_by_suima(nums[6], year),
                "te_wei": nums[6] % 10,
                "ping_nums": nums[:6],
            })
        except (ValueError, TypeError, KeyError):
            continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records


# ==================== 预测方案实现 ====================

def _build_models(records, up_to):
    """
    构建四个模型的数据结构
    返回: ta, tb, tc, td
    """
    ta = defaultdict(Counter)           # A: 本期特肖→下期特肖
    tb = defaultdict(lambda: defaultdict(float))  # B: 加权趋势
    tc = defaultdict(Counter)           # C: 二阶转移
    td = defaultdict(Counter)           # D: 号码→生肖

    for i in range(up_to - 1):
        ta[records[i]["te_sx"]][records[i + 1]["te_sx"]] += 1

    for i in range(up_to - 1):
        dist = up_to - 1 - i
        w = 1.5 if dist <= 12 else (1.2 if dist <= 24 else 1.0)
        tb[records[i]["te_sx"]][records[i + 1]["te_sx"]] += w

    for i in range(up_to - 2):
        tc[(records[i]["te_sx"], records[i + 1]["te_sx"])][records[i + 2]["te_sx"]] += 1

    for i in range(up_to - 1):
        td[records[i]["te_num"]][records[i + 1]["te_sx"]] += 1

    return ta, tb, tc, td


def _method_B_normal(records, up_to, ta, tb, tc, td):
    """
    [正常期] 四模型融合投票 + 杀本期特肖 + 平二num-9
    权重: A=0.30, B=0.30, C=0.20, D=0.20
    """
    cur_sx = records[up_to - 1]["te_sx"]
    prev_sx = records[up_to - 2]["te_sx"] if up_to >= 2 else cur_sx
    cur_num = records[up_to - 1]["te_num"]
    cur_year = records[up_to - 1]["year"]

    scores = Counter()

    # 模型A
    t = sum(ta[cur_sx].values())
    if t:
        for s, c in ta[cur_sx].items():
            scores[s] += (c / t) * 0.30

    # 模型B
    t = sum(tb[cur_sx].values())
    if t:
        for s, c in tb[cur_sx].items():
            scores[s] += (c / t) * 0.30

    # 模型C
    key = (prev_sx, cur_sx)
    if key in tc:
        t = sum(tc[key].values())
        if t:
            for s, c in tc[key].items():
                scores[s] += (c / t) * 0.20

    # 模型D
    t = sum(td[cur_num].values())
    if t:
        for s, c in td[cur_num].items():
            scores[s] += (c / t) * 0.20

    # 杀1: 杀本期特肖
    if cur_sx in scores:
        del scores[cur_sx]

    # 杀2: 平二号码-9 → 查当年生肖
    if len(records[up_to - 1]["ping_nums"]) >= 2:
        p2_num = records[up_to - 1]["ping_nums"][1]
        kill_sx = get_shengxiao_by_suima(offset_num(p2_num, -9), cur_year)
        if kill_sx in scores:
            del scores[kill_sx]

    # 取Top9
    top9 = [s for s, _ in scores.most_common(9)]
    if len(top9) < 9:
        freq = Counter(r["te_sx"] for r in records[:up_to])
        for s, _ in freq.most_common(12):
            if s not in top9:
                top9.append(s)
            if len(top9) >= 9:
                break
    return set(top9[:9])


def _method_wei3_BD_reverse(records, up_to, ta, tb, tc, td):
    """
    [尾3特征] 模型B反向 + 模型D反向 + A正常 + C正常 + 杀本期特肖 + num9
    权重: A=0.30, B=-0.30, C=0.20, D=-0.20
    触发期内命中率: 80.5%
    """
    cur_sx = records[up_to - 1]["te_sx"]
    prev_sx = records[up_to - 2]["te_sx"] if up_to >= 2 else cur_sx
    cur_num = records[up_to - 1]["te_num"]
    cur_year = records[up_to - 1]["year"]

    scores = Counter()

    # A: 正常
    t = sum(ta[cur_sx].values())
    if t:
        for s, c in ta[cur_sx].items():
            scores[s] += (c / t) * 0.30

    # B: 反向
    t = sum(tb[cur_sx].values())
    if t:
        for s, c in tb[cur_sx].items():
            scores[s] -= (c / t) * 0.30

    # C: 正常
    key = (prev_sx, cur_sx)
    if key in tc:
        t = sum(tc[key].values())
        if t:
            for s, c in tc[key].items():
                scores[s] += (c / t) * 0.20

    # D: 反向
    t = sum(td[cur_num].values())
    if t:
        for s, c in td[cur_num].items():
            scores[s] -= (c / t) * 0.20

    # 杀本期特肖
    if cur_sx in scores:
        del scores[cur_sx]

    # 平二num-9
    if len(records[up_to - 1]["ping_nums"]) >= 2:
        p2_num = records[up_to - 1]["ping_nums"][1]
        kill_sx = get_shengxiao_by_suima(offset_num(p2_num, -9), cur_year)
        if kill_sx in scores:
            del scores[kill_sx]

    top9 = [s for s, _ in scores.most_common(9)]
    if len(top9) < 9:
        freq = Counter(r["te_sx"] for r in records[:up_to])
        for s, _ in freq.most_common(12):
            if s not in top9:
                top9.append(s)
            if len(top9) >= 9:
                break
    return set(top9[:9])


def _method_wei7_sx_plus6(records, up_to):
    """
    [尾7特征] 本期特肖 +6 → 该生肖为中心 ±4 → 9肖
    触发期内命中率: 78.9%
    """
    cur_sx = records[up_to - 1]["te_sx"]
    cur_idx = ZODIAC.index(cur_sx)
    center_idx = (cur_idx + 6) % 12
    return sx_window(ZODIAC[center_idx])


def _method_pig_te7(records, up_to):
    """
    [上期猪特征] 本期特码+7 → ±4窗口号码 → 查当年生肖 → 9肖集合
    触发期内命中率: 80.8%
    """
    te_num = records[up_to - 1]["te_num"]
    year = records[up_to - 1]["year"]
    center = offset_num(te_num, 7)
    return set(
        get_shengxiao_by_suima(offset_num(center, w), year)
        for w in range(-4, 5)
    )


def _method_long_pig(records, up_to):
    """
    [本期龙特征] 以"猪"生肖为中心 ±4 → 9肖
    触发期内命中率: 79.0%
    """
    return sx_window("猪")


# ==================== 主预测函数 ====================

def predict_9xiao(records, up_to):
    """
    核心九肖预测函数
    参数:
        records: 历史开奖记录列表（由extract_records生成）
        up_to: 当前期索引（用records[:up_to]的历史数据预测第up_to期）
    返回:
        set: 预测的9个生肖
    """
    # 冷启动：数据不足时返回随机9肖
    if up_to < 50:
        return set(ZODIAC[:9])

    # 获取本期特征
    cur_wei = records[up_to - 1]["te_wei"]
    cur_sx = records[up_to - 1]["te_sx"]
    prev_sx = records[up_to - 2]["te_sx"] if up_to >= 2 else None

    # 构建四模型（所有方案共享）
    ta, tb, tc, td = _build_models(records, up_to)

    # 优先级: 尾3 > 尾7 > 上期猪 > 本期龙

    # 特征1: 尾3 → B反 + D反
    if cur_wei == 3:
        return _method_wei3_BD_reverse(records, up_to, ta, tb, tc, td)

    # 特征2: 尾7 → 特肖+6 ±4
    if cur_wei == 7:
        return _method_wei7_sx_plus6(records, up_to)

    # 特征3: 上期猪 → 特码+7 ±4
    if prev_sx == "猪":
        return _method_pig_te7(records, up_to)

    # 特征4: 本期龙 → 猪±4
    if cur_sx == "龙":
        return _method_long_pig(records, up_to)

    # 正常期 → 方法B
    return _method_B_normal(records, up_to, ta, tb, tc, td)


def predict_9xiao_latest():
    """
    便捷函数：自动加载最新数据，预测下期九肖
    返回:
        dict: {
            "latest_issue": 最新期号,
            "latest_code": 最新开奖号码,
            "latest_te_sx": 最新特肖,
            "predicted_9xiao": 预测的9个生肖列表,
            "features_triggered": 触发的特征（正常/尾3/尾7/上期猪/本期龙）
        }
    """
    data = load_all_data(auto_update=False)
    records = extract_records(data)

    if len(records) < 50:
        return {"error": "数据不足"}

    latest = records[-1]
    cur_wei = latest["te_wei"]
    cur_sx = latest["te_sx"]
    prev_sx = records[-2]["te_sx"] if len(records) >= 2 else None

    # 判断特征
    feature = "正常"
    if cur_wei == 3:
        feature = "尾3"
    elif cur_wei == 7:
        feature = "尾7"
    elif prev_sx == "猪":
        feature = "上期猪"
    elif cur_sx == "龙":
        feature = "本期龙"

    # 预测
    pred = predict_9xiao(records, len(records))

    # 获取最新开奖信息
    all_nums = []
    try:
        oc = data[-1].get("openCode", "")
        all_nums = [int(p.strip()) for p in oc.split(",")] if oc else []
    except:
        pass

    return {
        "latest_issue": latest["qishu"],
        "latest_code": ",".join(str(n) for n in all_nums) if all_nums else "",
        "latest_te_sx": cur_sx,
        "latest_te_wei": cur_wei,
        "predicted_9xiao": sorted(pred, key=lambda x: ZODIAC.index(x)),
        "features_triggered": feature,
    }


# ==================== 回测函数（用于验证） ====================

def backtest_full(records, min_history=200):
    """
    完整回测：逐期预测并统计命中率和连错分布
    返回: (命中率, 结果列表, 连错分布)
    """
    results = []
    for idx in range(min_history, len(records)):
        pred = predict_9xiao(records, idx)
        actual = records[idx]["te_sx"]
        results.append(actual in pred)

    hits = sum(results)
    total = len(results)
    rate = hits / total * 100 if total > 0 else 0

    max_streak = cur = 0
    streak_dist = Counter()
    for r in results:
        if not r:
            cur += 1
        else:
            if cur > 0:
                streak_dist[cur] += 1
            cur = 0
    if cur > 0:
        streak_dist[cur] += 1
    if streak_dist:
        max_streak = max(streak_dist.keys())

    return rate, results, streak_dist, max_streak
