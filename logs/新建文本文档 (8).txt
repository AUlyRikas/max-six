xiao48.py — 按本期特肖 / 按特肖合数 分组统计动态杀肖命中率与连错

import os, sys
from collections import defaultdict, Counter

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
MARK6_DIR = os.path.join(BASE_DIR, "mark6") if os.path.exists(os.path.join(BASE_DIR, "mark6")) else BASE_DIR
if MARK6_DIR not in sys.path: sys.path.insert(0, MARK6_DIR)

from shuju_loader import load_all_data
from shx_suishu import get_shengxiao_by_suima, SHENGXIAO

ZODIAC = SHENGXIAO
POS_NAMES = ['平一', '平二', '平三', '平四', '平五', '平六', '特码']

def offset_num(num, offset): return (num-1+offset)%49+1

def heshu(num):
    """号码合数 1~13"""
    s = num//10 + num%10
    r = s % 13
    return 13 if r==0 else r

def extract(data):
    records = []
    for item in data:
        try:
            qs=str(item.get("expect","")); oc=str(item.get("openCode",""))
            ot=item.get("openTime",""); year=int(ot[:4]) if ot else (int(qs[:4]) if len(qs)>=4 else 2026)
            if not qs or not oc: continue
            parts=oc.strip().split(",")
            if len(parts)!=7: continue
            nums=[int(p.strip()) for p in parts]
            records.append({
                "qishu":qs, "year":year,
                "te_num":nums[6], "te_sx":get_shengxiao_by_suima(nums[6],year),
                "te_wei":nums[6]%10, "te_he":heshu(nums[6]),
                "ping_nums":nums[:6],
                "ping_sx":[get_shengxiao_by_suima(n,year) for n in nums[:6]],
            })
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records

def build_l3_for_group(records, up_to, filter_fn):
    """为满足filter_fn的历史期建立L3规则库"""
    stats=defaultdict(lambda:{"total":0,"hit":0})
    for i in range(up_to-1):
        if not filter_fn(records, i): continue
        curr, nxt = records[i], records[i+1]
        year = curr["year"]; nxt_sx = nxt["te_sx"]
        for pos_idx, pos_name in enumerate(POS_NAMES):
            num = curr["ping_nums"][pos_idx] if pos_idx<6 else curr["te_num"]
            sx = get_shengxiao_by_suima(num, year)
            for off in range(12):
                new_num = offset_num(num, off); new_sx = get_shengxiao_by_suima(new_num, year)
                key = (pos_name, off, sx, new_sx)
                stats[key]["total"] += 1
                if new_sx != nxt_sx: stats[key]["hit"] += 1
    rules = []
    for (pos, off, sx, killed_sx), v in stats.items():
        if v["total"] >= 5:  # 放宽样本门槛
            rules.append({
                "pos":pos, "offset":off, "ping_sx":sx, "killed_sx":killed_sx,
                "samples":v["total"], "kill_rate":round(v["hit"]/v["total"]*100,2)
            })
    rules.sort(key=lambda x: x["kill_rate"], reverse=True)
    return rules

def match_best_rule(rules, records, idx):
    """本期7个位置匹配最高命中率规则"""
    year = records[idx]["year"]
    best = None
    for pos_idx, pos_name in enumerate(POS_NAMES):
        num = records[idx]["ping_nums"][pos_idx] if pos_idx<6 else records[idx]["te_num"]
        sx = get_shengxiao_by_suima(num, year)
        for r in rules:
            if r["pos"]==pos_name and r["ping_sx"]==sx:
                if best is None or r["kill_rate"]>best["kill_rate"] or (r["kill_rate"]==best["kill_rate"] and r["samples"]>best["samples"]):
                    best = r
    return best

def main():

    print("="*65)
    data = load_all_data(auto_update=False); records = extract(data)

        filter_fn = lambda r, i: r[i]["te_sx"] == sx
        rules = build_l3_for_group(records, len(records), filter_fn)

        # 回测该生肖触发期
        kills = total = 0
        max_streak = cur_streak = 0
        streak_dist = Counter()

        for idx in range(min_hist, len(records)):
            if records[idx-1]["te_sx"] != sx: continue
            total += 1
            # 滚动：只用历史数据
            rolling_rules = build_l3_for_group(records, idx, filter_fn)
            best = match_best_rule(rolling_rules, records, idx-1)
            if best:
                killed = best["killed_sx"]
                if records[idx]["te_sx"] != killed:
                    kills += 1
                    cur_streak = 0
                else:
                    cur_streak += 1
                    max_streak = max(max_streak, cur_streak)
                    if cur_streak == 1: streak_dist[1] = streak_dist.get(1,0)+1
            else:
                cur_streak = 0

        # 找最优规则
        best_rules = rules[:3] if rules else []
        best_str = " | ".join(f"{r['pos']}【{r['ping_sx']}】+{r['offset']}→{r['killed_sx']} {r['kill_rate']}% n={r['samples']}" for r in best_rules)

def analyze_zodiac(records, target_sx, min_hist=200):
    """分析本期特肖=target_sx时的七位置杀肖表现"""
    
    # 建立规则库（只用历史数据中特肖=target_sx的期）
    stats = defaultdict(lambda: {"total":0, "hit":0, "result_sx":None})
    for i in range(len(records)-1):
        if records[i]["te_sx"] != target_sx: continue
        curr, nxt = records[i], records[i+1]
        year = curr["year"]; nxt_sx = nxt["te_sx"]
        for pos_idx, pos_name in enumerate(POS_NAMES):
            num = curr["ping_nums"][pos_idx] if pos_idx<6 else curr["te_num"]
            sx = get_shengxiao_by_suima(num, year)
            for off in range(12):
                new_num = offset_num(num, off); new_sx = get_shengxiao_by_suima(new_num, year)
                key = (pos_name, off, sx)
                stats[key]["total"] += 1
                stats[key]["result_sx"] = new_sx
                if new_sx != nxt_sx: stats[key]["hit"] += 1

    # 每个位置找最优规则
    results = {}
    for pos_name in POS_NAMES:
        pos_rules = []
        for (pn, off, sx), v in stats.items():
            if pn != pos_name or v["total"] < 5: continue
            pos_rules.append({
                "ping_sx": sx, "offset": off, "killed_sx": v["result_sx"],
                "samples": v["total"], "kill_rate": round(v["hit"]/v["total"]*100,2)
            })
        pos_rules.sort(key=lambda x: x["kill_rate"], reverse=True)
        results[pos_name] = pos_rules[:5]  # 每位置Top5
    
    # 回测：每期取该位置最优规则杀肖，统计杀对率和连错
    pos_kill_stats = {}
    for pos_name in POS_NAMES:
        kills = total = 0
        max_streak = cur = 0
        for idx in range(min_hist, len(records)):
            if records[idx-1]["te_sx"] != target_sx: continue
            total += 1
            year = records[idx-1]["year"]
            # 滚动规则
            rolling_stats = defaultdict(lambda: {"total":0, "hit":0})
            for i in range(idx-1):
                if records[i]["te_sx"] != target_sx: continue
                curr, nxt = records[i], records[i+1]
                y = curr["year"]; nxt_sx = nxt["te_sx"]
                pos_idx = POS_NAMES.index(pos_name)
                num = curr["ping_nums"][pos_idx] if pos_idx<6 else curr["te_num"]
                sx = get_shengxiao_by_suima(num, y)
                for off in range(12):
                    new_num = offset_num(num, off); new_sx = get_shengxiao_by_suima(new_num, y)
                    key = (off, sx)
                    rolling_stats[key]["total"] += 1
                    if new_sx != nxt_sx: rolling_stats[key]["hit"] += 1
            # 本期匹配
            pos_idx = POS_NAMES.index(pos_name)
            num = records[idx-1]["ping_nums"][pos_idx] if pos_idx<6 else records[idx-1]["te_num"]
            sx = get_shengxiao_by_suima(num, year)
            best_rate = 0; best_killed = None
            for (off, psx), v in rolling_stats.items():
                if psx == sx and v["total"] >= 3:
                    rate = v["hit"]/v["total"]
                    if rate > best_rate:
                        best_rate = rate; best_killed = get_shengxiao_by_suima(offset_num(num, off), year)
            if best_killed:
                if records[idx]["te_sx"] != best_killed:
                    kills += 1; cur = 0
                else:
                    cur += 1; max_streak = max(max_streak, cur)
        pos_kill_stats[pos_name] = {
            "total": total, "kills": kills,
            "rate": round(kills/total*100,2) if total>0 else 0,
            "max_streak": max_streak
        }
    
    return results, pos_kill_stats

# -*- coding: utf-8 -*-
"""

import os, sys
from collections import defaultdict, Counter

BASE_DIR = os.path.dirname(os.path.abspath(__file__))

from shuju_loader import load_all_data
from shx_suishu import get_shengxiao_by_suima, SHENGXIAO

ZODIAC = SHENGXIAO
POS_NAMES = ['平一', '平二', '平三', '平四', '平五', '平六', '特码']

def offset_num(num, offset): return (num-1+offset)%49+1

def extract(data):
    records = []
    for item in data:
        try:
            qs=str(item.get("expect","")); oc=str(item.get("openCode",""))
            ot=item.get("openTime",""); year=int(ot[:4]) if ot else (int(qs[:4]) if len(qs)>=4 else 2026)
            if not qs or not oc: continue
            parts=oc.strip().split(",")
            if len(parts)!=7: continue
            nums=[int(p.strip()) for p in parts]
            records.append({
                "qishu":qs, "year":year,
                "te_num":nums[6], "te_sx":get_shengxiao_by_suima(nums[6],year),
                "ping_nums":nums[:6],
                "ping_sx":[get_shengxiao_by_suima(n,year) for n in nums[:6]],
            })
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records

def analyze_zodiac_full(records, target_sx, min_samples=5, min_rate=93.0, min_hist=200):
    """返回所有≥min_rate的规则及其滚动回测连错"""
    
    # 建立规则库
    stats = defaultdict(lambda: {"total":0, "hit":0, "result_sx":None})
    for i in range(len(records)-1):
        if records[i]["te_sx"] != target_sx: continue
        curr, nxt = records[i], records[i+1]
        year = curr["year"]; nxt_sx = nxt["te_sx"]
        for pos_idx, pos_name in enumerate(POS_NAMES):
            num = curr["ping_nums"][pos_idx] if pos_idx<6 else curr["te_num"]
            sx = get_shengxiao_by_suima(num, year)
            for off in range(12):
                new_num = offset_num(num, off); new_sx = get_shengxiao_by_suima(new_num, year)
                key = (pos_name, off, sx)
                stats[key]["total"] += 1
                stats[key]["result_sx"] = new_sx
                if new_sx != nxt_sx: stats[key]["hit"] += 1

    # 筛选≥min_rate的规则
    qualified = []
    for (pos_name, off, sx), v in stats.items():
        if v["total"] < min_samples: continue
        rate = v["hit"]/v["total"]*100
        if rate >= min_rate:
            qualified.append({
                "pos": pos_name, "offset": off, "ping_sx": sx,
                "killed_sx": v["result_sx"], "samples": v["total"],
                "kill_rate": round(rate, 2)
            })
    qualified.sort(key=lambda x: x["kill_rate"], reverse=True)

    # 对每条规则做滚动回测统计连错
    for rule in qualified:
        max_streak = cur = 0
        streak_dist = Counter()
        for idx in range(min_hist, len(records)):
            if records[idx-1]["te_sx"] != target_sx: continue
            pos_idx = POS_NAMES.index(rule["pos"])
            year = records[idx-1]["year"]
            num = records[idx-1]["ping_nums"][pos_idx] if pos_idx<6 else records[idx-1]["te_num"]
            sx = get_shengxiao_by_suima(num, year)
            if sx != rule["ping_sx"]: continue  # 本期该位置生肖不匹配，跳过
            
            killed = get_shengxiao_by_suima(offset_num(num, rule["offset"]), year)
            if records[idx]["te_sx"] == killed:
                cur += 1
                max_streak = max(max_streak, cur)
            else:
                if cur > 0: streak_dist[cur] = streak_dist.get(cur, 0) + 1
                cur = 0
        if cur > 0: streak_dist[cur] = streak_dist.get(cur, 0) + 1
        rule["max_streak"] = max_streak
        rule["streak_detail"] = dict(streak_dist) if streak_dist else {}
    
    return qualified

def main():

    print(f"\n===== xiao50.py 完成 =====")
