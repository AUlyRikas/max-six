#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
xiao6.py — 特码位置 + 全位置 双维度规则画像
输出：位置 | 最佳偏移 | 全局命中率 | 极值命中率 | 触发条件 | 所得生肖 | 样本
"""
import os
import sys
from collections import defaultdict, Counter

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
if BASE_DIR not in sys.path:
    sys.path.insert(0, BASE_DIR)

from shuju_loader import load_all_data

ZODIAC = ["鼠", "牛", "虎", "兔", "龙", "蛇", "马", "羊", "猴", "鸡", "狗", "猪"]
YEAR_SX = {2020: "鼠", 2021: "猪", 2022: "狗", 2023: "鸡", 2024: "猴", 2025: "羊", 2026: "马"}


def get_sx(num, year):
    start = YEAR_SX.get(year, "鼠")
    idx = (ZODIAC.index(start) - (num - 1)) % 12
    return ZODIAC[idx]


def offset_num(num, offset):
    return (num - 1 + offset) % 49 + 1


def extract(data):
    records = []
    for item in data:
        try:
            qs = str(item.get("expect", ""))
            oc = str(item.get("openCode", ""))
            year = int(qs[:4]) if len(qs) >= 4 else item.get("_year", 2020)
            if not qs or not oc: continue
            parts = oc.strip().split(",")
            if len(parts) != 7: continue
            nums = [int(p.strip()) for p in parts]
            records.append({
                "qishu": qs, "year": year,
                "te_num": nums[6], "te_sx": get_sx(nums[6], year),
                "ping_nums": nums[:6],
                "ping_sx": [get_sx(n, year) for n in nums[:6]],
            })
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records


def scan_position(records, pos, min_samples=30):
    """
    扫描单个位置：
    1. 找到全局最优偏移
    2. 在该偏移下找到极值命中率的条件
    """
    pos_names = ["平一", "平二", "平三", "平四", "平五", "平六", "特码"]

    # 全局：每个偏移的命中率
    offset_stats = defaultdict(lambda: {"kill": 0, "total": 0})
    # 双条件：每个(偏移, 生肖)的命中率
    detail_stats = defaultdict(lambda: {"kill": 0, "total": 0, "result_sx": None})

    for i in range(len(records) - 1):
        year = records[i]["year"]
        if pos < 6:
            num = records[i]["ping_nums"][pos]
            sx = records[i]["ping_sx"][pos]
        else:
            num = records[i]["te_num"]
            sx = records[i]["te_sx"]
        next_sx = records[i + 1]["te_sx"]

        for offset in range(-11, 12):
            new_num = offset_num(num, offset)
            result_sx = get_sx(new_num, year)
            offset_stats[offset]["total"] += 1
            if next_sx != result_sx:
                offset_stats[offset]["kill"] += 1

            key = (offset, sx)
            detail_stats[key]["total"] += 1
            detail_stats[key]["result_sx"] = result_sx
            if next_sx != result_sx:
                detail_stats[key]["kill"] += 1

    # 找全局最优偏移
    best_offset = None
    best_global_rate = 0
    for offset, data in offset_stats.items():
        if data["total"] >= 63:  # 样本≥63（与图片一致）
            rate = data["kill"] / data["total"] * 100
            if rate > best_global_rate:
                best_global_rate = rate
                best_offset = offset

    # 在该最优偏移下找极值条件
    best_detail = None
    best_detail_rate = 0
    for (offset, sx), data in detail_stats.items():
        if offset == best_offset and data["total"] >= min_samples:
            rate = data["kill"] / data["total"] * 100
            if rate > best_detail_rate:
                best_detail_rate = rate
                best_detail = {
                    "sx": sx,
                    "result_sx": data["result_sx"],
                    "samples": data["total"],
                    "rate": rate,
                }

    return {
        "pos": pos_names[pos],
        "best_offset": best_offset,
        "global_rate": best_global_rate,
        "detail": best_detail,
    }

        else:

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
xiao7.py — L3动态位移杀肖回测（含特码位置）
每期7个位置各取最优偏移杀1肖 → 取命中率最高的3个杀掉
剩下9肖 → 检查是否包含下期特肖
严格滚动：每期只用该期之前的数据统计规则
"""
import os
import sys
from collections import defaultdict

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
if BASE_DIR not in sys.path:
    sys.path.insert(0, BASE_DIR)

from shuju_loader import load_all_data
from shx_suishu import get_shengxiao_by_suima, SHENGXIAO

POS_NAMES = ['平一', '平二', '平三', '平四', '平五', '平六', '特码']


def offset_num(num, offset):
    new_num = num + offset
    if new_num > 49: new_num -= 49
    elif new_num < 1: new_num += 49
    return new_num


def extract(data):
    records = []
    for item in data:
        try:
            qs = str(item.get("expect", ""))
            oc = str(item.get("openCode", ""))
            ot = item.get("openTime", "")
            year = int(ot[:4]) if ot else (int(qs[:4]) if len(qs) >= 4 else 2026)
            if not qs or not oc: continue
            parts = oc.strip().split(",")
            if len(parts) != 7: continue
            nums = [int(p.strip()) for p in parts]
            records.append({
                "qishu": qs, "year": year,
                "te_num": nums[6],
                "te_sx": get_shengxiao_by_suima(nums[6], year),
                "ping_nums": nums[:6],
            })
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records


def build_l3_rules(records, up_to, min_samples=30):
    """
    统计 records[:up_to] 中 L3 动态位移规则
    返回: rules列表，每条包含 位置,偏移,平码生肖,所得生肖,样本量,命中率(杀对率)
    """
    stats = defaultdict(lambda: {"total": 0, "hit": 0})

    for i in range(up_to - 1):
        curr, nxt = records[i], records[i + 1]
        year = curr["year"]
        nxt_year = nxt["year"]
        nxt_sx = get_shengxiao_by_suima(nxt["te_num"], nxt_year)

        # 六个平码 + 特码
        for pos_idx, pos_name in enumerate(POS_NAMES):
            if pos_idx < 6:
                num = curr["ping_nums"][pos_idx]
            else:
                num = curr["te_num"]
            sx = get_shengxiao_by_suima(num, year)

            for offset_int in range(1, 12):
                for sign in [1, -1]:
                    offset_str = f"{'+' if sign > 0 else '-'}{offset_int}"
                    new_num = offset_num(num, sign * offset_int)
                    new_sx = get_shengxiao_by_suima(new_num, year)
                    key = (pos_name, offset_str, sx, new_sx)
                    stats[key]["total"] += 1
                    if new_sx != nxt_sx:
                        stats[key]["hit"] += 1

    rules = []
    for (pos, offset, sx, killed_sx), v in stats.items():
        if v["total"] >= min_samples:
            rules.append({
                "pos": pos,
                "offset": offset,
                "ping_sx": sx,
                "killed_sx": killed_sx,
                "samples": v["total"],
                "kill_rate": round(v["hit"] / v["total"] * 100, 2),
            })
    return rules


def match_and_kill(records, up_to, rules):
    """
    本期7个位置各匹配最优规则 → 取命中率最高的3个杀掉
    返回: (杀掉的3个生肖列表, 剩余9肖列表, 命中率列表, 低命中率标记列表)
    """
    year = records[up_to - 1]["year"]

    # 7个位置各自最优杀肖
    best_per_pos = []
    for pos_idx, pos_name in enumerate(POS_NAMES):
        if pos_idx < 6:
            num = records[up_to - 1]["ping_nums"][pos_idx]
        else:
            num = records[up_to - 1]["te_num"]
        sx = get_shengxiao_by_suima(num, year)

        # 找该位置该生肖下命中率最高的规则
        best_rule = None
        for r in rules:
            if r["pos"] == pos_name and r["ping_sx"] == sx:
                if best_rule is None or r["kill_rate"] > best_rule["kill_rate"] or (
                    r["kill_rate"] == best_rule["kill_rate"] and r["samples"] > best_rule["samples"]
                ):
                    best_rule = r
        if best_rule:
            best_per_pos.append(best_rule)

    if not best_per_pos:
        return [], [], [], []

    # 按命中率排序，取最高的3个
    best_per_pos.sort(key=lambda x: x["kill_rate"], reverse=True)
    top3 = best_per_pos[:3]

    killed = [r["killed_sx"] for r in top3]
    kill_rates = [r["kill_rate"] for r in top3]

    # 去重（两个位置可能杀同一生肖）
    killed_unique = list(dict.fromkeys(killed))
    kill_rates_unique = []
    for sx in killed_unique:
        for r in top3:
            if r["killed_sx"] == sx:
                kill_rates_unique.append(r["kill_rate"])
                break

    # 低命中率标记（<80%）
    low_flags = [r["kill_rate"] < 80 for r in top3]

    # 剩余9肖
    remaining = [sx for sx in SHENGXIAO if sx not in killed_unique]

    return killed_unique[:3], remaining[:9], kill_rates_unique[:3], low_flags[:3]


def main():
    print("xiao7.py — L3动态杀肖回测（取Top3杀掉，剩9肖）")
    print("=" * 60)

    data = load_all_data(auto_update=False)
    records = extract(data)
    print(f"记录数: {len(records)}")

    min_hist = 200

    # 回测
    hits_9xiao = 0
    total = 0
    low_count = 0
    miss_detail = []  # 记录未命中期的详情

    for idx in range(min_hist, len(records)):
        rules = build_l3_rules(records, idx, min_samples=30)
        killed, remaining, rates, lows = match_and_kill(records, idx, rules)

        if len(remaining) < 9:
            continue

        actual_sx = records[idx]["te_sx"]
        is_hit = actual_sx in remaining

        if is_hit:
            hits_9xiao += 1
        else:
            miss_detail.append({
                "issue": records[idx]["qishu"],
                "killed": killed,
                "rates": rates,
                "actual": actual_sx,
            })

        if any(lows):
            low_count += 1
        total += 1

        if total % 500 == 0:
            rate = hits_9xiao / total * 100
            print(f"  {total:4d}期: {rate:.2f}% ({hits_9xiao}/{total})")

    final_rate = hits_9xiao / total * 100 if total > 0 else 0
    print(f"\n最终命中率: {final_rate:.2f}% ({hits_9xiao}/{total})")
    print(f"低命中率标记(<80%)出现: {low_count}/{total} 期 ({low_count/total*100:.1f}%)")

    # 未命中期的误杀分析
    if miss_detail:
        print(f"\n未命中 {len(miss_detail)} 期，最近10期详情:")
        print(f"{'期号':<10} {'杀掉':<20} {'命中率':<20} {'实际特肖'}")
        print("-" * 70)
        for m in miss_detail[-10:]:
            killed_str = "、".join(m["killed"])
            rates_str = "、".join(f"{r}%" for r in m["rates"])
            print(f"{m['issue']:<10} {killed_str:<20} {rates_str:<20} {m['actual']}")

    # 对比现有九肖
    print(f"\n{'='*60}")
    print("对比:")
    print(f"  L3动态杀肖取Top3 → 剩9肖: {final_rate:.2f}%")
    print(f"  原九肖(四模型+杀本期特肖): 76.02%")
    print(f"  原九肖+num-9: 76.43%")
    print(f"  随机基线(9/12): 75.00%")

    print(f"\n===== xiao7.py 完成 =====")


if __name__ == "__main__":
    main()

    if up_to < 50: return ZODIAC[:9]
    ta = defaultdict(Counter)
    tb = defaultdict(lambda: defaultdict(float))
    tc = defaultdict(Counter)
    td = defaultdict(Counter)
    for i in range(up_to - 1):
        ta[records[i]["te_sx"]][records[i+1]["te_sx"]] += 1
    for i in range(up_to - 1):
        dist = up_to - 1 - i
        w = 1.5 if dist <= 12 else (1.2 if dist <= 24 else 1.0)
        tb[records[i]["te_sx"]][records[i+1]["te_sx"]] += w
    for i in range(up_to - 2):
        tc[(records[i]["te_sx"], records[i+1]["te_sx"])][records[i+2]["te_sx"]] += 1
    for i in range(up_to - 1):
        td[records[i]["te_num"]][records[i+1]["te_sx"]] += 1
    cur_sx = records[up_to-1]["te_sx"]
    prev_sx = records[up_to-2]["te_sx"] if up_to >= 2 else cur_sx
    cur_num = records[up_to-1]["te_num"]
    scores = Counter()
    t = sum(ta[cur_sx].values())
    if t:
        for s, c in ta[cur_sx].items(): scores[s] += (c/t)*0.30
    t = sum(tb[cur_sx].values())
    if t:
        for s, c in tb[cur_sx].items(): scores[s] += (c/t)*0.30
    t = sum(tc[(prev_sx, cur_sx)].values())
    if t:
        for s, c in tc[(prev_sx, cur_sx)].items(): scores[s] += (c/t)*0.20
    t = sum(td[cur_num].values())
    if t:
        for s, c in td[cur_num].items(): scores[s] += (c/t)*0.20
    if cur_sx in scores: del scores[cur_sx]
    top9 = [s for s, _ in scores.most_common(9)]
    if len(top9) < 9:
        freq = Counter(r["te_sx"] for r in records[:up_to])
        for s, _ in freq.most_common(12):
            if s not in top9: top9.append(s)
            if len(top9) >= 9: break
    return top9[:9]

def method_C_te7(records, up_to):
    """特码+7 ±4窗口"""
    if up_to < 1: return ZODIAC[:9]
    te_num = records[up_to-1]["te_num"]
    year = records[up_to-1]["year"]
    center = offset_num(te_num, +7)
    sx_set = set()
    for w in range(-4, 5):
        n = offset_num(center, w)
        sx_set.add(get_shengxiao_by_suima(n, year))
    return list(sx_set)[:9]


def backtest_with_streak(records, method_fn, name, min_hist=200):
    results = []
    for idx in range(min_hist, len(records)):
        pred = method_fn(records, idx)
        results.append(records[idx]["te_sx"] in pred)
    
    hits = sum(results)
    total = len(results)
    rate = hits / total * 100
    
    # 连错分析
    max_streak = 0
    cur_streak = 0
    streak_dist = Counter()
    for r in results:
        if not r:
            cur_streak += 1
        else:
            if cur_streak > 0:
                streak_dist[cur_streak] += 1
            cur_streak = 0
    if cur_streak > 0:
        streak_dist[cur_streak] += 1
    max_streak = max(streak_dist.keys()) if streak_dist else 0
    
    print(f"  {name}: {rate:.2f}% ({hits}/{total}) 最大连错{max_streak}期")
    if streak_dist:
        for k in sorted(streak_dist.keys()):
            print(f"    连错{k}期: {streak_dist[k]}次")
    return rate, max_streak
