def get_kill_rates(records, up_to):
    """7位置最优杀肖规则的杀对率，返回{生肖: 杀对率}"""

                key=(pos_name,off,sx,new_sx)

                if new_sx!=nxt_sx: stats[key]["hit"]+=1
    # 本期匹配最优规则

    kill_rates={}

        num=records[up_to-1]["ping_nums"][pos_idx] if pos_idx<6 else records[up_to-1]["te_num"]
        sx=get_shengxiao_by_suima(num,year)
        best=None
        for (pn,off,psx,ksx),v in stats.items():

        if best: kill_rates[best["killed"]]=max(kill_rates.get(best["killed"],0),best["rate"])

盆栽理论 · 讨论快照与代码推进方向
快照日期：2026-05-10
核心理论：生肖系统存在自我平衡机制，热肖到达阈值后回调，冷肖到达阈值后反弹，中间可能经过温肖作为“踏板”。

一、核心假设
假设	描述
热肖回调	当某个生肖在近N期内高频出现（热肖），它在下一期的出现概率会下降
冷肖反弹	当某个生肖在近N期内极少出现（冷肖），它在下一期的出现概率会上升
温肖踏板	热肖回调时，不一定直接转冷，可能先经过温区（振荡衰减）
阈值	存在一个出现频率的临界点，超过之后回调/反弹的概率显著增加
二、当前数据基础
资源	状态
历史数据	2180期（2020~2026，已去重）
九肖命中率	76.11%（四大模型滚动验证）
动态位移杀对率	95.73%（最大连错2期）
双条件规则	51条（四种类型、样本≥30）
三、可验证方向
方向A：热肖回调阈值扫描
核心问题：近10期中某个生肖出现次数达到多少时，下一期出现概率显著低于随机基线？

方法：

对每期，统计12生肖在近10期的出现次数

按频次分组（出现2次、3次、4次...）

统计各组在下一期的实际出现概率

找出概率显著跌破 1/12（8.33%）的频次阈值

如果发现：近10期出现≥4次的生肖，下一期出现概率只有5%，那么这个4次就是“热肖回调触发线”。

方向B：冷肖反弹阈值扫描
核心问题：某个生肖连续多少期未出现后，反弹概率显著高于随机？

方法：

统计每个生肖的连续遗漏期数

按遗漏期数分组（遗漏5期、10期、15期...）

统计各组在下一期的实际出现概率

找出概率显著高于 8.33% 的遗漏阈值

如果发现：连续遗漏≥12期的生肖，下一期出现概率达到14%，那么这个12期就是“冷肖反弹临界点”。

方向C：温肖踏板效应验证
核心问题：热肖在回调过程中，是否倾向于先进入“温区”（出现频率中等）再逐步降至冷区，还是直接跳转到冷区？

方向A: 近N期出现频率高的生肖(热肖) → 降权
方向B: 连续遗漏期数多的生肖(冷肖) → 加权
"""
import os, sys
from collections import defaultdict, Counter

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
MARK6_DIR = os.path.join(BASE_DIR, "mark6") if os.path.exists(os.path.join(BASE_DIR, "mark6")) else BASE_DIR

ZODIAC = SHENGXIAO

def offset_num(num, offset): return (num-1+offset)%49+1

def extract(data):
    records = []
    for item in data:
        try:
            qs=str(item.get("expect","")); oc=str(item.get("openCode",""))
            ot=item.get("openTime",""); year=int(ot[:4]) if ot else (int(qs[:4]) if len(qs)>=4 else 2026)
            if not qs or not oc: continue
            parts=oc.strip().split(",")
            if len(parts)!=7: continue
            nums=[int(p.strip()) for p in parts]
            records.append({"qishu":qs,"year":year,"te_num":nums[6],"te_sx":get_shengxiao_by_suima(nums[6],year),"te_wei":nums[6]%10,"ping_nums":nums[:6]})
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))

def get_9xiao_pool(records, up_to):

    cur_wei=records[up_to-1]["te_wei"]; cur_sx=records[up_to-1]["te_sx"]

        idx=ZODIAC.index('猪')

    ta=defaultdict(Counter); tb=defaultdict(lambda:defaultdict(float))
    tc=defaultdict(Counter); td=defaultdict(Counter)
    for i in range(up_to-1): ta[records[i]["te_sx"]][records[i+1]["te_sx"]]+=1
    for i in range(up_to-1):
        dist=up_to-1-i; w=1.5 if dist<=12 else (1.2 if dist<=24 else 1.0)

def get_cold_hot_scores(records, up_to, window=30):
    """冷热度得分：热肖降权，冷肖加权"""
    # 近window期内每个生肖的出现次数
    hot_count = Counter()
    start = max(0, up_to - window)
    for i in range(start, up_to):
        hot_count[records[i]["te_sx"]] += 1
    
    # 连续遗漏期数
    missing_streak = {}
    for sx in ZODIAC:
        streak = 0
        for i in range(up_to - 1, -1, -1):
            if records[i]["te_sx"] != sx:
                streak += 1
            else:
                break
        missing_streak[sx] = streak
    
    # 综合得分：热肖(出现多)→负分，冷肖(遗漏多)→正分
    scores = Counter()
    max_hot = max(hot_count.values()) if hot_count else 1
    max_miss = max(missing_streak.values()) if missing_streak else 1
    
    for sx in ZODIAC:
        hot_penalty = hot_count.get(sx, 0) / max_hot  # 0~1, 越热越大
        cold_bonus = missing_streak[sx] / max_miss     # 0~1, 越冷越大
        scores[sx] = cold_bonus - hot_penalty
    
    return scores

def main():

    print("="*55)
    data=load_all_data(auto_update=False); records=extract(data)

    for window in [10, 20, 30, 50]:
        res=[]
        for idx in range(min_hist, len(records)):
            pool=get_9xiao_pool(records, idx)
            cold_hot=get_cold_hot_scores(records, idx, window=window)
            
            # 池内按冷热度排序
            pool_scores={s: cold_hot.get(s,0) for s in pool}
            sorted_pool=sorted(pool_scores.items(), key=lambda x: x[1], reverse=True)
            top6=set(s for s,_ in sorted_pool[:6])
            res.append(records[idx]["te_sx"] in top6)

        rate=sum(res)/len(res)*100
        max_s=cur=0; sd=Counter()
        for r in res:
            if not r: cur+=1
            else:
                if cur>0: sd[cur]+=1
                cur=0
        if cur>0: sd[cur]+=1
        if sd: max_s=max(sd.keys())
        over3=sum(v for k,v in sd.items() if k>=3)
        print(f"窗口{window:3d}期: {rate:.2f}% 最大连错{max_s}期 连错≥3:{over3}次")

    print(f"\n随机基线(6/12): 50.00%")
    print(f"===== xiao46.py 完成 =====")

if __name__=="__main__":
    main()

def combined_sort(records, up_to, w_trans=0.4, w_cold=0.3, w_kill=0.3, cold_window=30):
    """三信号综合排序"""
    cur_sx=records[up_to-1]["te_sx"]; cur_year=records[up_to-1]["year"]
    
    # 信号1: 模型A转移频率（本期特肖→下期特肖）
    ta=defaultdict(Counter)
    for i in range(up_to-1): ta[records[i]["te_sx"]][records[i+1]["te_sx"]]+=1
    trans_scores=Counter()
    if cur_sx in ta:
        t=sum(ta[cur_sx].values())
        for s,c in ta[cur_sx].items(): trans_scores[s]=c/t
    
    # 信号2: 冷热度（冷加权，热降权）
    hot=Counter()
    for i in range(max(0,up_to-cold_window), up_to): hot[records[i]["te_sx"]]+=1
    missing={}
    for sx in ZODIAC:
        streak=0
        for i in range(up_to-1,-1,-1):
            if records[i]["te_sx"]!=sx: streak+=1
            else: break
        missing[sx]=streak
    max_hot=max(hot.values()) if hot else 1
    max_miss=max(missing.values()) if missing else 1
    cold_scores=Counter()
    for sx in ZODIAC: cold_scores[sx]=missing[sx]/max_miss - hot.get(sx,0)/max_hot
    
    # 信号3: 动态杀对率反向（杀对率高→该生肖不该来→降权）
    stats=defaultdict(lambda:{"total":0,"hit":0})
    for i in range(up_to-1):
        curr,nxt=records[i],records[i+1]; year=curr["year"]; nxt_sx=nxt["te_sx"]
        for pos_idx,pos_name in enumerate(POS_NAMES):
            num=curr["ping_nums"][pos_idx] if pos_idx<6 else curr["te_num"]
            sx=get_shengxiao_by_suima(num,year)
            for off in range(12):
                new_num=offset_num(num,off); new_sx=get_shengxiao_by_suima(new_num,year)
                key=(pos_name,off,sx,new_sx)
                stats[key]["total"]+=1
                if new_sx!=nxt_sx: stats[key]["hit"]+=1
    kill_rates={}
    for pos_idx,pos_name in enumerate(POS_NAMES):
        num=records[up_to-1]["ping_nums"][pos_idx] if pos_idx<6 else records[up_to-1]["te_num"]
        sx=get_shengxiao_by_suima(num,cur_year)
        best=None
        for (pn,off,psx,ksx),v in stats.items():
            if pn==pos_name and psx==sx and v["total"]>=30:
                if best is None or v["hit"]/v["total"]>best["rate"]:
                    best={"killed":ksx,"rate":v["hit"]/v["total"]}
        if best: kill_rates[best["killed"]]=max(kill_rates.get(best["killed"],0),best["rate"])
    kill_scores=Counter()
    for sx in ZODIAC: kill_scores[sx]=1-kill_rates.get(sx,0.5)
    
    # 归一化后加权合并
    for sc in [trans_scores, cold_scores, kill_scores]:
        vals=list(sc.values())
        mx=max(vals) if vals else 1; mn=min(vals) if vals else 0
        if mx>mn:
            for s in sc: sc[s]=(sc[s]-mn)/(mx-mn)
    
    final=Counter()
    for sx in ZODIAC:
        final[sx]=trans_scores.get(sx,0)*w_trans + cold_scores.get(sx,0)*w_cold + kill_scores.get(sx,0)*w_kill
    return final

                top6=set(s for s,_ in sorted_pool[:6])
                res.append(records[idx]["te_sx"] in top6)
            rate=sum(res)/len(res)*100

    print(f"\n最优: 转移{best_params[0]:.1f} 冷热{best_params[1]:.1f} 杀反{best_params[2]:.1f} → {best_rate:.2f}%")
    print(f"随机基线: 50.00%")
