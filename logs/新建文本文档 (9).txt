def query_kill(current_sx, position, trigger_sx):
    """查询杀肖规则"""
    if current_sx not in SHAXIAO_RULES: return []
    if position not in SHAXIAO_RULES[current_sx]: return []
    return SHAXIAO_RULES[current_sx][position].get(trigger_sx, [])

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
build_rules_json.py — 从 xiao50_output.txt 自动提取规则库
输出: 特肖杀肖规则库.json
"""
import json, os, re

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
MARK6_DIR = os.path.join(BASE_DIR, "mark6") if os.path.exists(os.path.join(BASE_DIR, "mark6")) else BASE_DIR

input_path = os.path.join(MARK6_DIR, "xiao50_output.txt")
output_path = os.path.join(BASE_DIR, "特肖杀肖规则库.json")

SHAXIAO_RULES = {}
current_sx = None

with open(input_path, 'r', encoding='utf-8') as f:
    for line in f:
        # 匹配生肖标题: 【马】共160期
        m = re.match(r'【(.+?)】共\d+期', line)
        if m:
            current_sx = m.group(1)
            SHAXIAO_RULES[current_sx] = {}
            continue
        
        if not current_sx: continue
        
        # 匹配规则行: 平一     +0   狗      →狗     100.0%    n=15   0期      无
        m = re.match(r'(平[一二三四五六]|特码)\s+([+-]?\d+)\s+(\S+)\s+→(\S+)\s+([\d.]+)%\s+n=(\d+)\s+(\d+)期', line)
        if m:
            pos = m.group(1)
            offset = int(m.group(2))
            trigger = m.group(3)
            kill = m.group(4)
            rate = float(m.group(5))
            sample = int(m.group(6))
            streak = int(m.group(7))
            
            # 筛选: ≥95% 且 连错≤1
            if rate < 95.0 or streak > 1: continue
            
            if pos not in SHAXIAO_RULES[current_sx]:
                SHAXIAO_RULES[current_sx][pos] = {}
            if trigger not in SHAXIAO_RULES[current_sx][pos]:
                SHAXIAO_RULES[current_sx][pos][trigger] = []
            
            SHAXIAO_RULES[current_sx][pos][trigger].append((offset, kill, rate, sample, streak))

# 同位置同触发内按杀对率降序排列
for sx in SHAXIAO_RULES:
    for pos in SHAXIAO_RULES[sx]:
        for trigger in SHAXIAO_RULES[sx][pos]:
            SHAXIAO_RULES[sx][pos][trigger].sort(key=lambda x: x[2], reverse=True)

with open(output_path, 'w', encoding='utf-8') as f:
    json.dump(SHAXIAO_RULES, f, ensure_ascii=False, indent=2)

print(f"✅ 已保存: {output_path}")
print(f"   {len(SHAXIAO_RULES)}个生肖")
for sx in SHAXIAO_RULES:
    total_rules = sum(len(v) for pos in SHAXIAO_RULES[sx] for v in SHAXIAO_RULES[sx][pos].values())
    print(f"   {sx}: {len(SHAXIAO_RULES[sx])}个位置, {total_rules}条规则")
