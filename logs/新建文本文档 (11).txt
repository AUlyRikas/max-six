#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
xiao58.py — 保持xiao53完整框架，微调评分权重
"""
import json, os, sys
from collections import defaultdict, Counter

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
MARK6_DIR = os.path.join(BASE_DIR, "mark6") if os.path.exists(os.path.join(BASE_DIR, "mark6")) else BASE_DIR
if MARK6_DIR not in sys.path: sys.path.insert(0, MARK6_DIR)

from shuju_loader import load_all_data
from shx_suishu import get_shengxiao_by_suima, SHENGXIAO

ZODIAC = SHENGXIAO

with open(os.path.join(BASE_DIR, "特肖杀肖规则库.json"), 'r', encoding='utf-8') as f:
    SHAXIAO_RULES = json.load(f)

def extract(data):
    records = []
    for item in data:
        try:
            qs=str(item.get("expect","")); oc=str(item.get("openCode",""))
            ot=item.get("openTime",""); year=int(ot[:4]) if ot else (int(qs[:4]) if len(qs)>=4 else 2026)
            if not qs or not oc: continue
            parts=oc.strip().split(",")
            if len(parts)!=7: continue
            nums=[int(p.strip()) for p in parts]
            records.append({
                "qishu":qs,"year":year,"te_num":nums[6],"te_sx":get_shengxiao_by_suima(nums[6],year),
                "te_wei":nums[6]%10,"ping_nums":nums[:6],
                "ping_sx":[get_shengxiao_by_suima(n,year) for n in nums[:6]],
            })
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records

def predict_six(records, idx, freq_weight, rule_weight):
    cur_sx = records[idx-1]["te_sx"]
    kills = set()
    rule_scores = Counter()
    
    if cur_sx in SHAXIAO_RULES:
        for pos_idx, pos_name in enumerate(["平一","平二","平三","平四","平五","平六","特码"]):
            if pos_name not in SHAXIAO_RULES[cur_sx]: continue
            actual_sx = records[idx-1]["ping_sx"][pos_idx] if pos_idx<6 else cur_sx
            if actual_sx in SHAXIAO_RULES[cur_sx][pos_name]:
                rules = SHAXIAO_RULES[cur_sx][pos_name][actual_sx]
                kills.add(rules[0][1])
                for r in rules:
                    rule_scores[r[1]] -= r[2]
    
    remaining = [s for s in ZODIAC if s not in kills]
    if len(remaining) < 9:
        freq = Counter(r["te_sx"] for r in records[:idx])
        for s, _ in freq.most_common(12):
            if s not in kills and s not in remaining:
                remaining.append(s)
            if len(remaining) >= 9: break
    
    nine = remaining[:9]
    freq = Counter(r["te_sx"] for r in records[:idx])
    max_f = max(freq.values()) if freq else 1
    
    final_scores = Counter()
    for s in nine:
        freq_score = freq.get(s, 0) / max_f * 100 * freq_weight
        rule_score = (rule_scores.get(s, 0) + 100) * rule_weight  # 归一化到正数
        final_scores[s] = freq_score + rule_score
    
    sorted_sx = sorted(final_scores.items(), key=lambda x: x[1], reverse=True)
    six = [s for s, _ in sorted_sx[:6]]
    return six

def backtest(records, freq_weight, rule_weight):
    min_hist = 200
    res = []
    for idx in range(min_hist, len(records)):
        six = predict_six(records, idx, freq_weight, rule_weight)
        res.append(records[idx]["te_sx"] in six)
    rate = sum(res)/len(res)*100
    max_s = cur = 0; sd = Counter()
    for r in res:
        if not r: cur += 1
        else:
            if cur > 0: sd[cur] += 1; cur = 0
    if cur > 0: sd[cur] += 1
    if sd: max_s = max(sd.keys())
    over3 = sum(v for k,v in sd.items() if k>=3)
    return rate, max_s, over3

def main():
    print("xiao58.py — 保持完整框架，微调权重")
    print("="*55)
    data = load_all_data(auto_update=False)
    records = extract(data)

    best_rate = 0; best_params = None
    results = []

    for freq_w in [0.3, 0.4, 0.5, 0.6, 0.7, 0.8]:
        rule_w = 1.0 - freq_w
        rate, ms, ov = backtest(records, freq_w, rule_w)
        results.append((rate, ms, ov, freq_w, rule_w))
        if rate > best_rate:
            best_rate = rate; best_params = (freq_w, rule_w)

    # 按命中率排序输出
    results.sort(key=lambda x: x[0], reverse=True)
    for rate, ms, ov, fw, rw in results:
        marker = "★" if rate == best_rate else " "
        print(f"  {marker} 频率{fw:.1f} 规则{rw:.1f}: {rate:.2f}% 最大连错{ms}期 连错≥3:{ov}次")

    print(f"\n最优: 频率{best_params[0]:.1f} 规则{best_params[1]:.1f} → {best_rate:.2f}%")
    print(f"原始(xiao53): 91.97% 连错2期9次")

if __name__ == "__main__":
    main()
