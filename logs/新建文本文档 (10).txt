#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
xiao51.py — 规则库杀肖映射表 筛选九肖
每期据本期特肖查表，7个位置各取最优规则杀1肖，去重后排除，剩9肖看命中率
"""
import json, os, sys
from collections import defaultdict, Counter

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
MARK6_DIR = os.path.join(BASE_DIR, "mark6") if os.path.exists(os.path.join(BASE_DIR, "mark6")) else BASE_DIR
if MARK6_DIR not in sys.path: sys.path.insert(0, MARK6_DIR)

from shuju_loader import load_all_data
from shx_suishu import get_shengxiao_by_suima, SHENGXIAO

ZODIAC = SHENGXIAO

# 加载规则库
RULES_PATH = os.path.join(BASE_DIR, "特肖杀肖规则库.json")
with open(RULES_PATH, 'r', encoding='utf-8') as f:
    SHAXIAO_RULES = json.load(f)

def extract(data):
    records = []
    for item in data:
        try:
            qs = str(item.get("expect", "")); oc = str(item.get("openCode", ""))
            ot = item.get("openTime", "")
            year = int(ot[:4]) if ot else (int(qs[:4]) if len(qs) >= 4 else 2026)
            if not qs or not oc: continue
            parts = oc.strip().split(",")
            if len(parts) != 7: continue
            nums = [int(p.strip()) for p in parts]
            records.append({
                "qishu": qs, "year": year,
                "te_num": nums[6], "te_sx": get_shengxiao_by_suima(nums[6], year),
                "te_wei": nums[6] % 10,
                "ping_nums": nums[:6],
                "ping_sx": [get_shengxiao_by_suima(n, year) for n in nums[:6]],
            })
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records

def get_kills_from_rules(records, idx):
    """根据本期特肖查表，7位置各取最优规则杀1肖"""
    cur_sx = records[idx-1]["te_sx"]
    if cur_sx not in SHAXIAO_RULES:
        return []
    
    kills = []
    pos_names = ["平一", "平二", "平三", "平四", "平五", "平六", "特码"]
    for pos_idx, pos_name in enumerate(pos_names):
        if pos_name not in SHAXIAO_RULES[cur_sx]:
            continue
        pos_rules = SHAXIAO_RULES[cur_sx][pos_name]
        actual_sx = records[idx-1]["ping_sx"][pos_idx] if pos_idx < 6 else cur_sx
        if actual_sx in pos_rules:
            # 取杀对率最高的一条
            best = pos_rules[actual_sx][0]
            kills.append(best[1])  # best[1] = 杀肖目标
    return kills

def main():
    print("xiao51.py — 规则库杀肖映射表 → 九肖")
    print("="*55)
    data = load_all_data(auto_update=False)
    records = extract(data)
    print(f"记录数: {len(records)}")
    min_hist = 200

    res = []
    kill_counts = Counter()
    remain_counts = Counter()

    for idx in range(min_hist, len(records)):
        kills = get_kills_from_rules(records, idx)
        killed_set = set(kills)
        remaining = [s for s in ZODIAC if s not in killed_set]
        remain_counts[len(remaining)] += 1
        kill_counts[len(killed_set)] += 1
        res.append(records[idx]["te_sx"] in remaining)

    rate = sum(res) / len(res) * 100
    print(f"\n命中率: {rate:.2f}% ({sum(res)}/{len(res)})")

    # 连错
    max_s = cur = 0
    sd = Counter()
    for r in res:
        if not r:
            cur += 1
        else:
            if cur > 0: sd[cur] += 1
            cur = 0
    if cur > 0: sd[cur] += 1
    if sd: max_s = max(sd.keys())
    over3 = sum(v for k, v in sd.items() if k >= 3)
    print(f"最大连错: {max_s}期  连错≥3: {over3}次")
    for k in sorted(sd.keys()):
        print(f"  连错{k}期: {sd[k]}次")

    print(f"\n杀肖分布: {dict(sorted(kill_counts.items()))}")
    print(f"剩余生肖分布: {dict(sorted(remain_counts.items()))}")
    print(f"九肖基线(方法B): 76.63%")

if __name__ == "__main__":
    main()

那么说说缺点和短板
93.49%不是九肖的命中率，实际剩余平均6.5肖，比九肖少了2.5个。

缺点
1. 覆盖范围不稳定。 每期剩余生肖数从5到10个不等，21%的期只剩5-6肖。范围太小的时候一旦触发连错，影响力被放大。

2. 规则库依赖历史统计。 7个位置各自查表，如果本期某个位置的平码生肖在规则库里没有≥95%的记录，那个位置就跳过了——杀的生肖少，剩余就多。反过来触发多的时候杀得狠。

3. 没有处理"规则冲突"。 两个位置可能指向同一个杀肖目标（去重后损失一条规则），也可能指向相互矛盾的杀肖建议。

4. 实盘验证不足。 这是查表回测，没有分训练集测试集。规则库本身是从全部2180期提取的，存在过拟合风险——规则库在它自己被提取的数据上表现好，新数据上可能打折扣。

短板
特码位置的规则覆盖最弱。 xiao50里特码位置很多生肖下都没有≥95%的规则（特码位置杀对率普遍比平一~平六低），导致实际7个位置中特码经常空过，只剩6个位置出规则。

那么怎么做优化是最优效果最好的,考虑资金和稳定性?
核心矛盾：剩余生肖数不稳定（5-10个），连错2期虽然只有5次，但剩余5-6肖时连错杀伤力比剩余9肖时大得多。

最优方案：剩余不足7肖时补高频生肖
规则库查表杀肖后，如果剩余少于7肖，用历史出现频率最高的生肖补足到7肖。这样既保留了规则库的高命中率（93.49%），又避免了范围过小带来的资金压力。

def get_kills_with_scores(records, idx):

    kills = set()

        if actual_sx in SHAXIAO_RULES[cur_sx][pos_name]:
            rules = SHAXIAO_RULES[cur_sx][pos_name][actual_sx]
            best = rules[0]
            kills.add(best[1])
            # 该规则的高杀对率→被杀生肖得低分，其他生肖不受影响
            # 反向：杀对率越低的规则，对应的"所得生肖"反而更可能

                # 杀对率越低→这个生肖越可能→加分

诊断
30次九肖命中但六肖未命中，全部排在第7名或第8名。 没有一次排第9名或更后。

这意味着九肖→六肖的排序只差临门一脚——实际特肖就在第7名，离Top6差一个身位。不是排序方法完全错误，是精度不够。当前评分排序（历史频率+规则反向得分）把正确答案推到了第7名边缘，就差一点点挤进前六。

解决方案
不是推翻排序，是在现有排序基础上加一个微调因子——当第6名和第7名分数差距极小（<某个阈值）时，把第7名也纳入六肖候选，随机舍弃第6或直接扩到七肖。或者引入第三个评分维度，把那23次第7名拉上来。

30次九肖命中但六肖未命中全是排第7名——卡在门槛上。当前评分排序里，这30次正确答案就差一口气。问题出在高频补足进来的那几个生肖占据了前6位置——它们是纯按历史频率排的，没有考虑当期规则库的信号。

改进方案：补足进来的生肖默认排在最后，不被杀且规则库触发的生肖优先。

python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
xiao55.py — 规则库杀肖 → 补9肖（补入排末位）→ 六肖
"""
import json, os, sys
from collections import defaultdict, Counter

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
MARK6_DIR = os.path.join(BASE_DIR, "mark6") if os.path.exists(os.path.join(BASE_DIR, "mark6")) else BASE_DIR
if MARK6_DIR not in sys.path: sys.path.insert(0, MARK6_DIR)

from shuju_loader import load_all_data
from shx_suishu import get_shengxiao_by_suima, SHENGXIAO

ZODIAC = SHENGXIAO

with open(os.path.join(BASE_DIR, "特肖杀肖规则库.json"), 'r', encoding='utf-8') as f:
    SHAXIAO_RULES = json.load(f)

def extract(data):
    records = []
    for item in data:
        try:
            qs=str(item.get("expect","")); oc=str(item.get("openCode",""))
            ot=item.get("openTime",""); year=int(ot[:4]) if ot else (int(qs[:4]) if len(qs)>=4 else 2026)
            if not qs or not oc: continue
            parts=oc.strip().split(",")
            if len(parts)!=7: continue
            nums=[int(p.strip()) for p in parts]
            records.append({
                "qishu":qs,"year":year,"te_num":nums[6],"te_sx":get_shengxiao_by_suima(nums[6],year),
                "te_wei":nums[6]%10,"ping_nums":nums[:6],
                "ping_sx":[get_shengxiao_by_suima(n,year) for n in nums[:6]],
            })
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records

def predict_six(records, idx):
    cur_sx = records[idx-1]["te_sx"]
    kills = set()
    native = []  # 规则库活下来的原生生肖
    filled = []  # 高频补进来的
    
    if cur_sx in SHAXIAO_RULES:
        for pos_idx, pos_name in enumerate(["平一","平二","平三","平四","平五","平六","特码"]):
            if pos_name not in SHAXIAO_RULES[cur_sx]: continue
            actual_sx = records[idx-1]["ping_sx"][pos_idx] if pos_idx<6 else cur_sx
            if actual_sx in SHAXIAO_RULES[cur_sx][pos_name]:
                kills.add(SHAXIAO_RULES[cur_sx][pos_name][actual_sx][0][1])
    
    for s in ZODIAC:
        if s not in kills:
            native.append(s)
    
    freq = Counter(r["te_sx"] for r in records[:idx])
    
    # 原生按频率排序
    native.sort(key=lambda s: freq.get(s, 0), reverse=True)
    
    # 取前9（先原生后补足标记）
    nine = native[:9]
    if len(nine) < 9:
        for s, _ in freq.most_common(12):
            if s not in kills and s not in nine:
                nine.append(s)
                filled.append(s)
            if len(nine) >= 9:
                break
    
    # 六肖：原生优先取前6，不足再取补入
    six = [s for s in nine if s not in filled][:6]
    if len(six) < 6:
        for s in nine:
            if s not in six:
                six.append(s)
            if len(six) >= 6:
                break
    
    return six, kills, nine

def main():
    print("xiao55.py — 原生优先排序 → 六肖")
    print("="*55)
    data = load_all_data(auto_update=False)
    records = extract(data)
    min_hist = 200
    total = len(records) - min_hist

    res_6 = []
    for idx in range(min_hist, len(records)):
        six, _, _ = predict_six(records, idx)
        res_6.append(records[idx]["te_sx"] in six)

    rate6 = sum(res_6) / total * 100
    max_s = cur = 0; sd = Counter()
    for r in res_6:
        if not r: cur += 1
        else:
            if cur > 0: sd[cur] += 1; cur = 0
    if cur > 0: sd[cur] += 1
    if sd: max_s = max(sd.keys())
    
    print(f"六肖命中率: {rate6:.2f}%")
    print(f"最大连错: {max_s}期  连错≥3: {sum(v for k,v in sd.items() if k>=3)}次")
    for k in sorted(sd.keys()): print(f"  连错{k}期: {sd[k]}次")
    print(f"\n原始版六肖: 91.97% 连错2期9次")

if __name__ == "__main__":
    main()
