def get_sx_nums(sx, year):

def build_rules_rolling(records, up_to):
    """

    rules = {}

                        rules[cur_sx][pn][actual_sx][offset]["last_error"], -999
                    )
                else:
                    # 记录连错
                    pass

    # 筛选杀对率≥95%且样本≥5的规则
    filtered = {}

python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
wei_dynamic_sx.py — 本期特肖→七位置位移→包含下期特肖 统计
逻辑：本期特肖=X时，7个位置各自的平码生肖做位移(±1~±11生肖位移)，
      得到的目标生肖=下期特肖的概率。
生肖用循环方法（12个生肖闭环），以本命年生肖为起始位置。
"""
import os, sys
from collections import defaultdict, Counter

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
MARK6_DIR = os.path.join(BASE_DIR, "mark6") if os.path.exists(os.path.join(BASE_DIR, "mark6")) else BASE_DIR
if MARK6_DIR not in sys.path:
    sys.path.insert(0, MARK6_DIR)

from shuju_loader import load_all_data
from shx_suishu import get_shengxiao_by_suima, SHENGXIAO, get_benming

ZODIAC = SHENGXIAO
POS_NAMES = ["平一", "平二", "平三", "平四", "平五", "平六", "特码"]


def shift_sx(sx, offset):
    """生肖循环位移，±1~±11"""
    idx = ZODIAC.index(sx)
    new_idx = (idx + offset) % 12
    return ZODIAC[new_idx]


def extract(data):
    records = []
    for item in data:
        try:
            qs = str(item.get("expect", ""))
            oc = str(item.get("openCode", ""))
            ot = item.get("openTime", "")
            year = int(ot[:4]) if ot else (int(qs[:4]) if len(qs) >= 4 else 2026)
            if not qs or not oc: continue
            parts = oc.strip().split(",")
            if len(parts) != 7: continue
            nums = [int(p.strip()) for p in parts]
            records.append({
                "qishu": qs, "year": year,
                "te_sx": get_shengxiao_by_suima(nums[6], year),
                "te_num": nums[6],
                "ping_sx": [get_shengxiao_by_suima(n, year) for n in nums[:6]],
                "ping_nums": nums[:6],
            })
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records


def main():
    print("wei_dynamic_sx.py — 本期特肖→七位置生肖位移→命中下期特肖")
    print("=" * 65)

    data = load_all_data(auto_update=False)
    records = extract(data)
    print(f"记录数: {len(records)}")
    min_hist = 200

    # 统计结构: stats[本期特肖][位置][位移量] = {"hit":次数, "total":次数}
    stats = defaultdict(lambda: defaultdict(lambda: defaultdict(lambda: {"hit": 0, "total": 0})))

    for idx in range(min_hist, len(records)):
        cur_sx = records[idx - 1]["te_sx"]
        nxt_sx = records[idx]["te_sx"]

        for pos_idx, pos_name in enumerate(POS_NAMES):
            ping_sx = records[idx - 1]["ping_sx"][pos_idx] if pos_idx < 6 else cur_sx

            for offset in range(-11, 12):
                if offset == 0: continue
                target_sx = shift_sx(ping_sx, offset)
                stats[cur_sx][pos_name][offset]["total"] += 1
                if target_sx == nxt_sx:
                    stats[cur_sx][pos_name][offset]["hit"] += 1

    # 输出每个本期特肖下，各位置的最优位移
    for sx in ZODIAC:
        print(f"\n{'='*65}")
        print(f"本期特肖: {sx}")
        print(f"{'位置':<6} {'最优位移':<8} {'命中率':<8} {'命中/总数':<10} {'目标生肖示例'}")
        print("-" * 55)

        for pos_name in POS_NAMES:
            best_offset = 0
            best_rate = 0
            best_hit = 0
            best_total = 0

            for offset in range(-11, 12):
                if offset == 0: continue
                d = stats[sx][pos_name][offset]
                if d["total"] >= 30:
                    rate = d["hit"] / d["total"] * 100
                    if rate > best_rate:
                        best_rate = rate
                        best_offset = offset
                        best_hit = d["hit"]
                        best_total = d["total"]

            if best_total > 0:
                sign = "+" if best_offset > 0 else ""
                example_sx = shift_sx(ZODIAC[0], best_offset)  # 示例：马+offset=?
                print(f"  {pos_name:<4}  {sign}{best_offset:<7} {best_rate:.2f}%    {best_hit}/{best_total:<8} 马{sign}{best_offset}={example_sx}")

    print(f"\n===== wei_dynamic_sx.py 完成 =====")


if __name__ == "__main__":
    main()

结论
生肖位移和号码位移是两套完全不同的逻辑。号码偏移走的是模49路线——号码变了，对应的年份生肖也跟着变，这就是MAX规则库能捕捉到信号的原因。生肖位移只做12生肖的循环平移，丢失了年份映射信息，所以你看到的就是纯粹随机分布。

这个数据反过来验证了另一个关键点：已知"本期特肖"这个信息本身不提供任何预测价值。不管你怎么平移本期的生肖，都和下期的特肖没有统计关系。

这个方向也到此为止了。尾数也好、生肖位移也好，纯统计偏移都撞在了随机墙上。MAX系统的核心价值还是在那套基于号码偏移+年份映射的规则库上。

六肖和九肖的排序规则：

九肖的排列依据
九肖是规则库杀肖后剩下的生肖 + 高频补足进来的生肖。排列顺序是：

先放规则库保留的原生生肖（按ZODIAC数组原始顺序），再放高频补足进来的生肖（按频率从高到低）。

也就是说九肖的排列不是按评分来的，是按"来源"分组的。原生在权，补足在后。

六肖的排序依据
六肖是从九肖池里按综合评分重新排序取Top6。评分公式：

text
得分 = 频率得分 × 0.3 + 规则反向得分 × 0.7

频率得分 = 该生肖历史出现次数 / 最高频生肖次数 × 100
规则反向得分 = (该生肖被规则库杀对率 + 100)
规则反向得分解释：

没被任何规则杀的生肖：100 × 0.7 = 70分

被一条95%规则杀的生肖：(100-95) × 0.7 = 3.5分

差距20倍。六肖的排序基本由"有没有被规则库盯上"决定。 频率只是微调——在所有"没被杀"的安全生肖里，历史上更热的排前面。

你看到的牛和虎
如果牛和虎都没被规则库杀，它们在六肖里排得靠前是合理的——它们确实是历史高频生肖。如果其中有一个被规则库杀了一部分位置但没全杀，得分会降低，排名会靠后。

你看不懂排序，是因为九肖和六肖用了两套不同的排列逻辑，可以统一一下：九肖也按同一套评分排序，这样九肖和六肖的排名就一致了，一眼就能看懂。

def extract_records(data):
    records = []
    for item in data:
        try:
            qs = str(item.get("expect", ""))
            oc = str(item.get("openCode", ""))
            ot = item.get("openTime", "")
            year = int(ot[:4]) if ot else (int(qs[:4]) if len(qs) >= 4 else 2026)
            if not qs or not oc: continue
            parts = oc.strip().split(",")
            if len(parts) != 7: continue
            nums = [int(p.strip()) for p in parts]
            records.append({
                "qishu": qs, "year": year,
                "te_num": nums[6], "te_sx": get_shengxiao_by_suima(nums[6], year),
                "te_wei": nums[6] % 10,
                "ping_nums": nums[:6],
                "ping_sx": [get_shengxiao_by_suima(n, year) for n in nums[:6]],
            })
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records


def predict_6xiao(records, up_to, freq_weight=0.3, rule_weight=0.7):
    cur_sx = records[up_to - 1]["te_sx"]
    kills = set()
    rule_scores = Counter()
    filled = []

    if cur_sx in SHAXIAO_RULES:
        for pos_idx, pos_name in enumerate(POS_NAMES):
            if pos_name not in SHAXIAO_RULES[cur_sx]: continue
            actual_sx = records[up_to - 1]["ping_sx"][pos_idx] if pos_idx < 6 else cur_sx
            if actual_sx in SHAXIAO_RULES[cur_sx][pos_name]:
                rules = SHAXIAO_RULES[cur_sx][pos_name][actual_sx]
                kills.add(rules[0][1])
                for r in rules: rule_scores[r[1]] -= r[2]

    remaining = [s for s in ZODIAC if s not in kills]
    if len(remaining) < 9:
        freq = Counter(r["te_sx"] for r in records[:up_to])
        for s, _ in freq.most_common(12):
            if s not in kills and s not in remaining:
                remaining.append(s)
                filled.append(s)
            if len(remaining) >= 9: break

    nine = remaining[:9]
    # === 新增：规则反向得分TOP3 ===
    border = sorted(rule_scores.items(), key=lambda x: x[1], reverse=True)
    top3 = set(s for s, _ in border[:3])
    # ============================
    freq = Counter(r["te_sx"] for r in records[:up_to])
    max_f = max(freq.values()) if freq else 1

    final_scores = Counter()
    for s in nine:
        freq_score = freq.get(s, 0) / max_f * 100 * freq_weight
        rule_score = (rule_scores.get(s, 0) + 100) * rule_weight
        final_scores[s] = freq_score + rule_score + (20 if s in top3 else 0)

    sorted_sx = sorted(final_scores.items(), key=lambda x: x[1], reverse=True)
    six = [s for s, _ in sorted_sx[:6]]
    return six, kills, nine, filled

def predict_direct(records, up_to, top_n=6):
    """直选版：规则反向得分+TOP加分，从12生肖直接排序取TopN"""
    cur_sx = records[up_to - 1]["te_sx"]
    rule_scores = Counter()

    if cur_sx in SHAXIAO_RULES:
        for pos_idx, pos_name in enumerate(POS_NAMES):
            if pos_name not in SHAXIAO_RULES[cur_sx]: continue
            actual_sx = records[up_to - 1]["ping_sx"][pos_idx] if pos_idx < 6 else cur_sx
            if actual_sx in SHAXIAO_RULES[cur_sx][pos_name]:
                rules = SHAXIAO_RULES[cur_sx][pos_name][actual_sx]
                for r in rules:
                    rule_scores[r[1]] -= r[2]

    border = sorted(rule_scores.items(), key=lambda x: x[1], reverse=True)
    top_bonus = set(s for s, _ in border[:top_n])

    freq = Counter(r["te_sx"] for r in records[:up_to])
    max_f = max(freq.values()) if freq else 1

    scores = Counter()
    for s in ZODIAC:
        freq_score = freq.get(s, 0) / max_f * 100 * 0.3
        rule_score = (rule_scores.get(s, 0) + 100) * 0.7
        scores[s] = freq_score + rule_score + (20 if s in top_bonus else 0)

    sorted_sx = sorted(scores.items(), key=lambda x: x[1], reverse=True)
    nine = [s for s, _ in sorted_sx[:9]]
    six = [s for s, _ in sorted_sx[:6]]

    # 规则杀肖（用于展示，不影响预测）
    kills = set()
    for s, sc in rule_scores.items():
        if sc < -90:
            kills.add(s)

    return six, nine, kills

    track["nine"].append(-1)
    track["six"].append(-1)

    if len(track["nine"]) > 100:

    return track

def extract_records(data):
    records = []
    for item in data:
        try:
            qs = str(item.get("expect", ""))
            oc = str(item.get("openCode", ""))
            ot = item.get("openTime", "")
            year = int(ot[:4]) if ot else (int(qs[:4]) if len(qs) >= 4 else 2026)
            if not qs or not oc: continue
            parts = oc.strip().split(",")
            if len(parts) != 7: continue

                "qishu": qs, "year": year,

                "te_wei": nums[6] % 10,
                "ping_nums": nums[:6],
                "ping_sx": [get_shengxiao_by_suima(n, year) for n in nums[:6]],
            })
        except: continue
    records.sort(key=lambda x: int(x["qishu"]))
    return records

def predict_direct(records, up_to, top_n=6):
    """直选版：规则反向得分+TOP加分，从12生肖直接排序取TopN"""
    cur_sx = records[up_to - 1]["te_sx"]
    rule_scores = Counter()

    if cur_sx in SHAXIAO_RULES:
        for pos_idx, pos_name in enumerate(POS_NAMES):
            if pos_name not in SHAXIAO_RULES[cur_sx]: continue
            actual_sx = records[up_to - 1]["ping_sx"][pos_idx] if pos_idx < 6 else cur_sx
            if actual_sx in SHAXIAO_RULES[cur_sx][pos_name]:
                rules = SHAXIAO_RULES[cur_sx][pos_name][actual_sx]
                for r in rules:
                    rule_scores[r[1]] -= r[2]

    border = sorted(rule_scores.items(), key=lambda x: x[1], reverse=True)
    top_bonus = set(s for s, _ in border[:top_n])

    freq = Counter(r["te_sx"] for r in records[:up_to])
    max_f = max(freq.values()) if freq else 1

    scores = Counter()
    for s in ZODIAC:
        freq_score = freq.get(s, 0) / max_f * 100 * 0.3
        rule_score = (rule_scores.get(s, 0) + 100) * 0.7
        scores[s] = freq_score + rule_score + (20 if s in top_bonus else 0)

    sorted_sx = sorted(scores.items(), key=lambda x: x[1], reverse=True)
    nine = [s for s, _ in sorted_sx[:9]]
    six = [s for s, _ in sorted_sx[:6]]

    # 规则杀肖（用于展示，不影响预测）
    kills = set()
    for s, sc in rule_scores.items():
        if sc < -90:
            kills.add(s)

    return six, nine, kills, rule_scores

def verify_last_prediction(records):
    """
    校验上一次预测是否命中。
    逻辑：读取track中最后一条记录，用records中对应期号的实际开奖对比。
    如果track最后一条的hit=-1（待验证），更新它。
    """
    track = load_hit_track()
    if not track:
        print("[命中追踪] 暂无历史预测记录，跳过校验")
        return track

    last = track[-1]
    if last.get("hit9", -1) != -1 and last.get("hit6", -1) != -1:
        # 已经校验过了
        return track

    # 找上期预测对应的实际开奖
    predicted_issue = last.get("issue", "")
    actual_sx = None
    for r in records:
        if r["qishu"] == predicted_issue:
            actual_sx = r["te_sx"]
            break

    if actual_sx is None:
        return track

    # 更新命中结果
    last["hit9"] = 1 if actual_sx in last.get("nine", []) else 0
    last["hit6"] = 1 if actual_sx in last.get("six", []) else 0
    track[-1] = last
    save_hit_track(track)

    hit9_str = "✓" if last["hit9"] else "✗"
    hit6_str = "✓" if last["hit6"] else "✗"
    print(f"[命中追踪] 上期{predicted_issue}已校验: 九肖{hit9_str} 六肖{hit6_str}")
    return track


def append_prediction_to_track(issue, nine, six):

        "hit6": -1,

    })
    # 只保留最近100条
    if len(track) > 100:
        track = track[-100:]
    save_hit_track(track)
    return track


def calc_dynamic_rate(window=50):
    """计算动态命中率"""
    track = load_hit_track()
    valid = [t for t in track if t.get("hit9", -1) >= 0][-window:]
    
    if not valid:
        return 0, 0, 0, 0
    
    hits9 = sum(t["hit9"] for t in valid)
    hits6 = sum(t["hit6"] for t in valid)
    total = len(valid)
    
    return hits9 / total * 100, hits6 / total * 100, hits9, hits6
